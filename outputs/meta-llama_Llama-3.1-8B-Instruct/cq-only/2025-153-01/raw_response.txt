:Ontology a owl:Ontology ;
  rdfs:label "Referent Transition Ontology" ;
  rdfs:comment "Modeling referent transitions and associated observations.".

  rdfs:label "Referent" ;
  rdfs:comment "An entity being referenced.".

  rdfs:label "Referent Transition" ;

  rdfs:range xsd:dateTime.

:ReferentTransitionTrigger a owl:Class ;
  rdfs:label "Referent Transition Trigger" ;
  rdfs:comment "A rule that triggered a referent transition.".

  rdfs:label "Temporal Binding Overlap" ;

  rdfs:label "Complete Context" ;
  rdfs:comment "A complete context for a time point.".
