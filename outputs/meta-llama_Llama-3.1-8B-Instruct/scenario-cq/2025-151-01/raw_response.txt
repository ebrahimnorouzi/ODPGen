### Ontology Design Pattern (ODP) in Turtle Syntax

```turtle
@prefix owl:  <http://www.w3.org/2002/07/owl#>.
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#>.
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#>.
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#>.
@prefix :     <http://example.org/odp#>.

:Ontology a owl:Ontology ;
  rdfs:label "Evaluation Request Ontology" ;
  rdfs:comment "Formal description of a requested action by an assignee on a target asset.".

:Request a :RequestClass ;
  rdfs:label "Request" ;
  rdfs:comment "Formal description of a requested action by an assignee on a target asset.".

:Action a owl:Class ;
  rdfs:label "Action" ;
  rdfs:comment "The requested action.".

:Assignee a owl:Class ;
  rdfs:label "Assignee" ;
  rdfs:comment "The party issuing the evaluation request.".

:Asset a owl:Class ;
  rdfs:label "Asset" ;
  rdfs:comment "The target asset of the evaluation request.".

:TemporalContext a owl:Class ;
  rdfs:label "Temporal Context" ;
  rdfs:comment "Temporal context of the evaluation request.".

:TemporalContext hasTemporalValue :TemporalValue ;
  rdfs:label "hasTemporalValue" ;
  rdfs:comment "Temporal value of the evaluation request." ;
  rdfs:domain :TemporalContext ;
  rdfs:range xsd:dateTime.

:TemporalValue a owl:DatatypeProperty ;
  rdfs:label "Temporal Value" ;
  rdfs:comment "Temporal value of the evaluation request." ;
  rdfs:domain :TemporalContext ;
  rdfs:range xsd:dateTime.

:SpatialContext a owl:Class ;
  rdfs:label "Spatial Context" ;
  rdfs:comment "Spatial context of the evaluation request.".

:EU a owl:Class ;
  rdfs:label "EU" ;
  rdfs:comment "Spatial context of the evaluation request (EU).".

:Role a owl:Class ;
  rdfs:label "Role" ;
  rdfs:comment "Role of the assignee.".

:RequestCapability a owl:Class ;
  rdfs:label "Request Capability" ;
  rdfs:comment "Request capability of the assignee.".

:RequestCapability hasCapability :Capability ;
  rdfs:label "hasCapability" ;
  rdfs:comment "Capability of the request capability." ;
  rdfs:domain :RequestCapability ;
  rdfs:range :Capability.

:Capability a owl:Class ;
  rdfs:label "Capability" ;
  rdfs:comment "Capability of the request capability.".

:RequestCapability hasEU :EU ;
  rdfs:label "hasEU" ;
  rdfs:comment "EU of the request capability." ;
  rdfs:domain :RequestCapability ;
  rdfs:range :EU.

:RequestCapability hasWeekday :Weekday ;
  rdfs:label "hasWeekday" ;
  rdfs:comment "Weekday of the request capability." ;
  rdfs:domain :RequestCapability ;
  rdfs:range :Weekday.

:Weekday a owl:Class ;
  rdfs:label "Weekday" ;
  rdfs:comment "Weekday of the request capability.".

:Request hasAction :Action ;
  rdfs:label "hasAction" ;
  rdfs:comment "The requested action." ;
  rdfs:domain :Request ;
  rdfs:range :Action.

:Request hasAssignee :Assignee ;
  rdfs:label "hasAssignee" ;
  rdfs:comment "The party issuing the evaluation request." ;
  rdfs:domain :Request ;
  rdfs:range :Assignee.

:Request hasAsset :Asset ;
  rdfs:label "hasAsset" ;
  rdfs:comment "The target asset of the evaluation request." ;
  rdfs:domain :Request ;
  rdfs:range :Asset.

:Request hasTemporalContext :TemporalContext ;
  rdfs:label "hasTemporalContext" ;
  rdfs:comment "Temporal context of the evaluation request." ;
  rdfs:domain :Request ;
  rdfs:range :TemporalContext.

:Request hasSpatialContext :SpatialContext ;
  rdfs:label "hasSpatialContext" ;
  rdfs:comment "Spatial context of the evaluation request." ;
  rdfs:domain :Request ;
  rdfs:range :SpatialContext.

:Request hasRole :Role ;
  rdfs
