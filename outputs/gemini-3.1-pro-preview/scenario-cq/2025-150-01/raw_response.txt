:Ontology a owl:Ontology ;
    rdfs:label "MQTT Process Data Ontology" ;
    rdfs:comment "An ontology design pattern for modeling MQTT message protocols, payloads, and their alignment with SOSA observation and actuation process descriptions in industrial and IoT settings." .

### Classes ###

:MQTTParticipant a owl:Class ;
    rdfs:label "MQTT Participant" ;
    rdfs:comment "A participant in an MQTT network, such as a Client or a Broker." .

:MQTTClient a owl:Class ;
    rdfs:subClassOf :MQTTParticipant ;
    rdfs:label "MQTT Client" ;
    rdfs:comment "An MQTT client that connects to a broker to publish or subscribe to topics." .

:MQTTBroker a owl:Class ;
    rdfs:subClassOf :MQTTParticipant ;
    rdfs:label "MQTT Broker" ;
    rdfs:comment "An MQTT broker that receives and distributes messages among connected clients." .

:MQTTNetworkConnection a owl:Class ;
    rdfs:label "MQTT Network Connection" ;
    rdfs:comment "A network connection established between MQTT participants." .

:MQTTTopic a owl:Class ;
    rdfs:label "MQTT Topic" ;
    rdfs:comment "An MQTT topic to which messages are published and subscribed." .

:MQTTTopicFilter a owl:Class ;
    rdfs:label "MQTT Topic Filter" ;
    rdfs:comment "An expression contained in a subscription to indicate an interest in one or more topics." .

:MQTTTopicSubject a owl:Class ;
    rdfs:label "MQTT Topic Subject" ;
    rdfs:comment "The semantic subject represented by an MQTT topic." .

:MQTTControlPacket a owl:Class ;
    rdfs:label "MQTT Control Packet" ;
    rdfs:comment "A packet of data sent across the network connection in MQTT." .

:MQTTPublishPacket a owl:Class ;
    rdfs:subClassOf :MQTTControlPacket ;
    rdfs:label "MQTT PUBLISH Packet" ;
    rdfs:comment "An MQTT control packet used to transport a message from a client to a broker or vice versa." .

:MQTTSubscribePacket a owl:Class ;
    rdfs:subClassOf :MQTTControlPacket ;
    rdfs:label "MQTT SUBSCRIBE Packet" ;
    rdfs:comment "An MQTT control packet used by a client to subscribe to one or more topics." .

:MQTTUnsubscribePacket a owl:Class ;
    rdfs:subClassOf :MQTTControlPacket ;
    rdfs:label "MQTT UNSUBSCRIBE Packet" ;
    rdfs:comment "An MQTT control packet used by a client to unsubscribe from one or more topics." .

:MQTTFixedHeader a owl:Class ;
    rdfs:label "MQTT Fixed Header" ;
    rdfs:comment "The fixed header present in all MQTT control packets." .

:MQTTVariableHeader a owl:Class ;
    rdfs:label "MQTT Variable Header" ;
    rdfs:comment "The variable header present in some MQTT control packets." .

:MQTTPayload a owl:Class ;
    rdfs:label "MQTT Payload" ;
    rdfs:comment "The payload part of an MQTT control packet." .

:MQTTSubscriptionEntry a owl:Class ;
    rdfs:label "MQTT Subscription Entry" ;
    rdfs:comment "An entry within a SUBSCRIBE payload detailing the topic filter and requested QoS." .

:MQTTPayloadField a owl:Class ;
    rdfs:label "MQTT Payload Field" ;
    rdfs:comment "A specific field encoded within an MQTT payload." .

:SosaPlatform a owl:Class ;
    rdfs:label "SOSA Platform" ;
    rdfs:comment "A platform that hosts SOSA systems, sensors, or actuators." .

:SosaSystem a owl:Class ;
    rdfs:label "SOSA System" ;
    rdfs:comment "A system in the SOSA ontology, superclass of sensors and actuators." .

:SosaSensor a owl:Class ;
    rdfs:subClassOf :SosaSystem ;
    rdfs:label "SOSA Sensor" ;
    rdfs:comment "A device, agent, or software that generates observations." .

:SosaActuator a owl:Class ;
    rdfs:subClassOf :SosaSystem ;
    rdfs:label "SOSA Actuator" ;
    rdfs:comment "A device that carries out actuations." .

:SosaFeatureOfInterest a owl:Class ;
    rdfs:label "SOSA Feature of Interest" ;
    rdfs:comment "The feature whose property is being observed or acted upon." .

:SosaProperty a owl:Class ;
    rdfs:label "SOSA Property" ;
    rdfs:comment "An observable or actuable quality of a Feature of Interest." .

:SosaObservation a owl:Class ;
    rdfs:label "SOSA Observation" ;
    rdfs:comment "An act of carrying out an observation procedure to estimate or calculate a value of a property." .

:SosaObservationCollection a owl:Class ;
    rdfs:label "SOSA Observation Collection" ;
    rdfs:comment "A collection of SOSA observations." .

:SosaActuation a owl:Class ;
    rdfs:label "SOSA Actuation" ;
    rdfs:comment "An act of carrying out an actuation procedure to change the state of the world." .

:SosaActuationCollection a owl:Class ;
    rdfs:label "SOSA Actuation Collection" ;
    rdfs:comment "A collection of SOSA actuations." .

:QUDTQuantityValue a owl:Class ;
    rdfs:label "QUDT Quantity Value" ;
    rdfs:comment "A quantity value expressing the magnitude and unit of a measure." .

:QUDTUnit a owl:Class ;
    rdfs:label "QUDT Unit" ;
    rdfs:comment "A unit of measurement." .
