:Ontology a owl:Ontology ;
    rdfs:label "Evaluation Request Ontology Design Pattern" ;
    rdfs:comment "An ontology design pattern representing evaluation requests, their issuers, target assets, requested actions, issuance time, and contextual constraints." .

### Classes ###

:EvaluationRequest a owl:Class ;
    rdfs:label "Evaluation Request" ;
    rdfs:comment "A formal request made by a party to evaluate a specific target asset, potentially constrained by contextual information." .

:Action a owl:Class ;
    rdfs:label "Action" ;
    rdfs:comment "The specific action that is requested to be performed or evaluated." .

:Party a owl:Class ;
    rdfs:label "Party" ;
    rdfs:comment "An entity (e.g., person, organization, or system) that issues the evaluation request." .

:Asset a owl:Class ;
    rdfs:label "Asset" ;
    rdfs:comment "The target resource, item, or entity that is the subject of the evaluation request." .

:ContextualInformation a owl:Class ;
    rdfs:label "Contextual Information" ;
    rdfs:comment "Additional data or constraints that provide context to the evaluation request and limit its scope." .

### Object Properties ###

:hasRequestedAction a owl:ObjectProperty ;
    rdfs:domain :EvaluationRequest ;
    rdfs:range :Action ;
    rdfs:label "has requested action" ;
    rdfs:comment "Links an evaluation request to the action that is being requested." .

:issuedBy a owl:ObjectProperty ;
    rdfs:domain :EvaluationRequest ;
    rdfs:range :Party ;
    rdfs:label "issued by" ;
    rdfs:comment "Links an evaluation request to the party that issued it." .

:hasTargetAsset a owl:ObjectProperty ;
    rdfs:domain :EvaluationRequest ;
    rdfs:range :Asset ;
    rdfs:label "has target asset" ;
    rdfs:comment "Links an evaluation request to the asset that is targeted for evaluation." .

:hasContextualInformation a owl:ObjectProperty ;
    rdfs:domain :EvaluationRequest ;
    rdfs:range :ContextualInformation ;
    rdfs:label "has contextual information" ;
    rdfs:comment "Links an evaluation request to additional contextual information or constraints." .

### Datatype Properties ###

:issuedAt a owl:DatatypeProperty ;
    rdfs:domain :EvaluationRequest ;
    rdfs:range xsd:dateTime ;
    rdfs:label "issued at" ;
    rdfs:comment "The date and time when the evaluation request was issued." .
