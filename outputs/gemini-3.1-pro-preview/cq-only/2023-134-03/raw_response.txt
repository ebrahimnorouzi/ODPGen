```turtle
@prefix owl:  <http://www.w3.org/2002/07/owl#> .
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#> .
@prefix :     <http://example.org/odp#> .

:Ontology a owl:Ontology ;
    rdfs:label "Registry Data Management Ontology Design Pattern" ;
    rdfs:comment "An OWL ontology design pattern to model registries, their maintainers, available data types, storage locations, and associated policies." .

### Classes ###

:Agent a owl:Class ;
    rdfs:label "Agent" ;
    rdfs:comment "An entity (such as a person, organization, or system) responsible for maintaining the registry." .

:Data a owl:Class ;
    rdfs:label "Data" ;
    rdfs:comment "A collection of information, dataset, or digital resource available within the registry." .

:DataType a owl:Class ;
    rdfs:label "Data Type" ;
    rdfs:comment "The specific category, format, or classification of the data." .

:Policy a owl:Class ;
    rdfs:label "Policy" ;
    rdfs:comment "A set of rules, guidelines, or licenses associated with the data governing its access and usage." .

:Registry a owl:Class ;
    rdfs:label "Registry" ;
    rdfs:comment "A centralized system or catalog that manages and provides access to data." .

:StorageLocation a owl:Class ;
    rdfs:label "Storage Location" ;
    rdfs:comment "The physical or virtual location (e.g., server, database, geographic region) where the data is stored." .

### Object Properties ###

:hasAvailableData a owl:ObjectProperty ;
    rdfs:label "has available data" ;
    rdfs:comment "Links a registry to the data it makes available to users." ;
    rdfs:domain :Registry ;
    rdfs:range :Data .

:hasDataType a owl:ObjectProperty ;
    rdfs:label "has data type" ;
    rdfs:comment "Links a data entity to its specific type or classification." ;
    rdfs:domain :Data ;
    rdfs:range :DataType .

:hasPolicy a owl:ObjectProperty ;
    rdfs:label "has policy" ;
    rdfs:comment "Links a data entity to the policy governing its use or access." ;
    rdfs:domain :Data ;
    rdfs:range :Policy .

:isMaintainedBy a owl:ObjectProperty ;
    rdfs:label "is maintained by" ;
    rdfs:comment "Links a registry to the agent responsible for its maintenance and upkeep." ;
    rdfs:domain :Registry ;
    rdfs:range :Agent .

:isStoredIn a owl:ObjectProperty ;
    rdfs:label "is stored in" ;
    rdfs:comment "Links a data entity to the location where it is physically or virtually stored." ;
    rdfs:domain :Data ;
    rdfs:range :StorageLocation .

### Datatype Properties ###

:dateCreated a owl:DatatypeProperty ;
    rdfs:label "date created" ;
    rdfs:comment "The date and time when the registry was initially created." ;
    rdfs:domain :Registry ;
    rdfs:range xsd:dateTime .

:dateUpdated a owl:DatatypeProperty ;
    rdfs:label "date updated" ;
    rdfs:comment "The date and time when the registry was last modified or updated." ;
    rdfs:domain :Registry ;
    rdfs:range xsd:dateTime .
```
