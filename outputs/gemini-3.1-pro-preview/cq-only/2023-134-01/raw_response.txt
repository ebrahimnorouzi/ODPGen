:Ontology a owl:Ontology ;

:Policy a owl:Class ;

    rdfs:comment "A set of rules or conditions governing access to personal data." .

    rdfs:label "Agent" ;
    rdfs:comment "An entity, such as a person or organization, that can create or be assigned a policy." .

:Application a owl:Class ;
    rdfs:label "Application" ;
    rdfs:comment "A software application or service used to access data." .

:AccessMode a owl:Class ;
    rdfs:label "Access Mode" ;
    rdfs:comment "The type of access requested or granted, such as read, write, or delete." .

:PersonalData a owl:Class ;
    rdfs:label "Personal Data" ;
    rdfs:comment "The specific personal data that is being accessed." .

:Purpose a owl:Class ;
    rdfs:label "Purpose" ;
    rdfs:comment "The reason or justification for accessing the personal data." .

:hasAssignee a owl:ObjectProperty ;

:accessesPersonalData a owl:ObjectProperty ;
    rdfs:label "accesses personal data" ;

    rdfs:range :PersonalData .

:hasPurpose a owl:ObjectProperty ;
    rdfs:label "has purpose" ;
    rdfs:comment "Specifies the purpose for accessing the data under the policy." ;
    rdfs:domain :Policy ;
    rdfs:range :Purpose .
