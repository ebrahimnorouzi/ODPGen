```turtle
@prefix owl:  <http://www.w3.org/2002/07/owl#> .
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#> .
@prefix :     <http://example.org/odp#> .

:Ontology a owl:Ontology ;
    rdfs:label "Causal Bayesian Network Ontology Design Pattern" ;
    rdfs:comment "An ontology design pattern for representing events, their states, and weighted causal relations (direct and mediated) in a Causal Bayesian Network." .

:Event a owl:Class ;
    rdfs:label "Event" ;
    rdfs:comment "A node in a Causal Bayesian Network representing an occurrence or condition." .

:EventState a owl:Class ;
    rdfs:label "Event State" ;
    rdfs:comment "The specific state or value of an event (e.g., on, off, wet, not wet)." .

:CausalRelation a owl:Class ;
    rdfs:label "Causal Relation" ;
    rdfs:comment "A directed edge in a Causal Bayesian Network representing a causal link between a cause event and an effect event." ;
    rdfs:subClassOf [
        a owl:Restriction ;
        owl:onProperty :hasCause ;
        owl:cardinality "1"^^xsd:nonNegativeInteger
    ] ,
    [
        a owl:Restriction ;
        owl:onProperty :hasEffect ;
        owl:cardinality "1"^^xsd:nonNegativeInteger
    ] .

:hasCause a owl:ObjectProperty ;
    rdfs:label "has cause" ;
    rdfs:comment "Links a causal relation to the event that acts as the cause." ;
    rdfs:domain :CausalRelation ;
    rdfs:range :Event .

:hasEffect a owl:ObjectProperty ;
    rdfs:label "has effect" ;
    rdfs:comment "Links a causal relation to the event that acts as the effect." ;
    rdfs:domain :CausalRelation ;
    rdfs:range :Event .

:hasMediator a owl:ObjectProperty ;
    rdfs:label "has mediator" ;
    rdfs:comment "Links a causal relation to an event that mediates the effect of the cause on the effect." ;
    rdfs:domain :CausalRelation ;
    rdfs:range :Event .

:hasState a owl:ObjectProperty ;
    rdfs:label "has state" ;
    rdfs:comment "Links an event to its current or possible state." ;
    rdfs:domain :Event ;
    rdfs:range :EventState .

:hasEffectWeight a owl:DatatypeProperty ;
    rdfs:label "has effect weight" ;
    rdfs:comment "The numerical weight or strength of the causal relation." ;
    rdfs:domain :CausalRelation ;
    rdfs:range xsd:float .
```
