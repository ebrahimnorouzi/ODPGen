:Ontology a owl:Ontology ;

  rdfs:comment "A minimal ontology design pattern for representing an author using alternative names, including pseudonyms and name changes, when publishing books." .

:Author a owl:Class ;
  rdfs:label "Author" ;
  rdfs:comment "A person in the role of writing books or other authored works." .

:NameUsage a owl:Class ;
  rdfs:label "Name usage" ;
  rdfs:comment "A contextual use of a name by an author, such as a pseudonym or a changed name used for publication." .

:Book a owl:Class ;
  rdfs:label "Book" ;
  rdfs:comment "A published book or book-like written work." .

:hasNameUsage a owl:ObjectProperty ;
  rdfs:label "has name usage" ;
  rdfs:comment "Relates an author to a name usage adopted by that author." ;
  rdfs:domain :Author ;
  rdfs:range :NameUsage .

:usedForPublication a owl:ObjectProperty ;
  rdfs:label "used for publication" ;
  rdfs:comment "Relates a name usage to a book published under that name." ;
  rdfs:domain :NameUsage ;
  rdfs:range :Book .

:legalNameOf a owl:ObjectProperty ;
  rdfs:label "legal name of" ;
  rdfs:comment "Relates a name usage to the author whose legal or personal naming history it belongs to." ;
  rdfs:domain :NameUsage ;
  rdfs:range :Author .

:nameString a owl:DatatypeProperty ;
  rdfs:label "name string" ;
  rdfs:comment "The literal textual form of a name usage." ;
  rdfs:domain :NameUsage ;
  rdfs:range xsd:string .

:bookTitle a owl:DatatypeProperty ;
  rdfs:label "book title" ;
  rdfs:comment "The literal title of a book." ;
  rdfs:domain :Book ;
  rdfs:range xsd:string .
