@prefix owl:  <http://www.w3.org/2002/07/owl#> .
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ndf-syntax-ns#> .

:Ontology a owl:Ontology ;
  rdfs:label "Cultural Heritage Survey ODP" ;
  rdfs:comment "A minimal ontology design pattern for representing cultural heritage survey activities, their targets, methods, observations, participants and roles, equipment configurations, interpretations, post-processed outputs, and photographic acquisition parameters." .

:SurveyActivity a owl:Class ;
  rdfs:label "Survey Activity" ;
  rdfs:comment "A scientific survey or investigation activity performed on a cultural property or a feature of interest." .

:CulturalProperty a owl:Class ;
  rdfs:label "Cultural Property" ;
  rdfs:comment "A cultural heritage entity that is the target of a survey or investigation activity." .

:FeatureOfInterest a owl:Class ;
  rdfs:label "Feature of Interest" ;
  rdfs:comment "A specific feature of a cultural property that is targeted by a survey or investigation activity." .

:Method a owl:Class ;
  rdfs:label "Method" ;
  rdfs:comment "An investigation method or technique adopted in a survey or investigation activity." .

:Observation a owl:Class ;
  rdfs:label "Observation" ;
  rdfs:comment "An observation produced during a survey or investigation activity." .

:Expert a owl:Class ;
  rdfs:label "Expert" ;
  rdfs:comment "A person participating in a survey or investigation activity." .

:Role a owl:Class ;
  rdfs:label "Role" ;
  rdfs:comment "The role with which an expert participates in a survey or investigation activity." .

:Participation a owl:Class ;
  rdfs:label "Participation" ;
  rdfs:comment "A participation relation linking an expert, a survey activity, and a role." .

:EquipmentConfiguration a owl:Class ;
  rdfs:label "Equipment Configuration" ;
  rdfs:comment "A hardware, software, and technical configuration used to conduct a survey or investigation activity." .

:Interpretation a owl:Class ;
  rdfs:label "Interpretation" ;
  rdfs:comment "An interpretation or claim supported by observations produced during a survey." .

:PostProcessedOutput a owl:Class ;
  rdfs:label "Post-Processed Output" ;
  rdfs:comment "An output such as an interactive image, spectrum, or dataset derived from an acquisition or observation activity." .

:PhotographicAcquisitionCycle a owl:Class ;
  rdfs:label "Photographic Acquisition Cycle" ;
  rdfs:comment "A photographic acquisition cycle occurring within a survey activity." ;
  rdfs:subClassOf :SurveyActivity .

:AcquisitionParameter a owl:Class ;
  rdfs:label "Acquisition Parameter" ;
  rdfs:comment "A parameter characterising a photographic acquisition cycle, such as exposure time, ISO, or aperture." .

:performedOnProperty a owl:ObjectProperty ;
  rdfs:label "performed on property" ;
  rdfs:comment "Relates a survey activity to the cultural property on which it was performed." ;
  rdfs:domain :SurveyActivity ;
  rdfs:range :CulturalProperty .

:performedOnFeature a owl:ObjectProperty ;
  rdfs:label "performed on feature" ;
  rdfs:comment "Relates a survey activity to the specific feature of interest on which it was performed." ;
  rdfs:domain :SurveyActivity ;
  rdfs:range :FeatureOfInterest .

:adoptedMethod a owl:ObjectProperty ;
  rdfs:label "adopted method" ;
  rdfs:comment "Relates a survey activity to the method or technique adopted in it." ;
  rdfs:domain :SurveyActivity ;
  rdfs:range :Method .

:producedObservation a owl:ObjectProperty ;
  rdfs:label "produced observation" ;
  rdfs:comment "Relates a survey activity to an observation produced during it." ;
  rdfs:domain :SurveyActivity ;
  rdfs:range :Observation .

:hasParticipation a owl:ObjectProperty ;
  rdfs:label "has participation" ;
  rdfs:comment "Relates a survey activity to a participation instance describing an expert and role in that activity." ;
  rdfs:domain :SurveyActivity ;
  rdfs:range :Participation .
