```turtle
@prefix owl:  <http://www.w3.org/2002/07/owl#> .
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#> .
@prefix :     <http://example.org/odp#> .

:Ontology a owl:Ontology ;
  rdfs:label "Survey and Investigation Activity ODP" ;
  rdfs:comment "An ontology design pattern for representing survey and investigation activities on cultural properties and their features, including methods, observations, participants and roles, equipment and configurations, interpretations supported by observations, post-processed outputs, and photographic acquisition parameters." .

:CulturalProperty a owl:Class ;
  rdfs:label "Cultural Property" ;
  rdfs:comment "A cultural property that can be the subject of a survey or investigation activity." .

:FeatureOfInterest a owl:Class ;
  rdfs:label "Feature of Interest" ;
  rdfs:comment "A specific feature, part, or aspect of a cultural property that can be the subject of a survey or investigation activity." .

:SurveyActivity a owl:Class ;
  rdfs:label "Survey Activity" ;
  rdfs:comment "A survey or investigation activity performed on a cultural property or on a specific feature of interest." .

:Method a owl:Class ;
  rdfs:label "Method" ;
  rdfs:comment "A method or investigation technique adopted in a survey or investigation activity." .

:Observation a owl:Class ;
  rdfs:label "Observation" ;
  rdfs:comment "An observation produced during a survey or investigation activity." .

:Expert a owl:Class ;
  rdfs:label "Expert" ;
  rdfs:comment "An expert who participates in a survey or investigation activity." .

:Role a owl:Class ;
  rdfs:label "Role" ;
  rdfs:comment "A role held by an expert in the context of participation in a survey or investigation activity." .

:Participation a owl:Class ;
  rdfs:label "Participation" ;
  rdfs:comment "A participation situation linking an expert to a survey or investigation activity and specifying the role held." .

:Hardware a owl:Class ;
  rdfs:label "Hardware" ;
  rdfs:comment "A hardware device used to conduct a survey or investigation activity." .

:Software a owl:Class ;
  rdfs:label "Software" ;
  rdfs:comment "A software application used to conduct a survey or investigation activity." .

:TechnicalConfiguration a owl:Class ;
  rdfs:label "Technical Configuration" ;
  rdfs:comment "A technical configuration used to conduct a survey or investigation activity." .

:Interpretation a owl:Class ;
  rdfs:label "Interpretation" ;
  rdfs:comment "An interpretation or claim supported by one or more observations." .

:PostProcessedOutput a owl:Class ;
  rdfs:label "Post-Processed Output" ;
  rdfs:comment "An output produced by post-processing, such as an interactive image, spectrum, or dataset, related to an acquisition or observation activity." .

:PhotographicAcquisitionCycle a owl:Class ;
  rdfs:label "Photographic Acquisition Cycle" ;
  rdfs:comment "A photographic acquisition cycle carried out within a survey or investigation activity." .

:AcquisitionParameter a owl:Class ;
  rdfs:label "Acquisition Parameter" ;
  rdfs:comment "A parameter characterising a photographic acquisition cycle, such as exposure time, ISO, or aperture." .

:ParameterType a owl:Class ;
  rdfs:label "Parameter Type" ;
  rdfs:comment "The type or kind of an acquisition parameter, such as exposure time, ISO, or aperture." .

:performedOnCulturalProperty a owl:ObjectProperty ;
  rdfs:label "performed on cultural property" ;
  rdfs:comment "Relates a survey or investigation activity to the cultural property on which it was performed." ;
  rdfs:domain :SurveyActivity ;
  rdfs:range :CulturalProperty .

:performedOnFeatureOfInterest a owl:ObjectProperty ;
  rdfs:label "performed on feature of interest" ;
  rdfs:comment "Relates a survey or investigation activity to the specific feature of interest on which it was performed." ;
  rdfs:domain :SurveyActivity ;
  rdfs:range :FeatureOfInterest .

:adoptedMethod a owl:ObjectProperty ;
  rdfs:label "adopted method" ;
  rdfs:comment "Relates a survey or investigation activity to the method or investigation technique adopted." ;
  rdfs:domain :SurveyActivity ;
