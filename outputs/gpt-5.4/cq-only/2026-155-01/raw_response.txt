```turtle
@prefix owl:  <http://www.w3.org/2002/07/owl#> .
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#> .
@prefix :     <http://example.org/odp#> .

:Ontology a owl:Ontology ;
    rdfs:label "Survey and Investigation Activity ODP" ;
    rdfs:comment "An ontology design pattern for representing survey and investigation activities on cultural properties and their features, including methods, observations, participants and roles, equipment and configurations, interpretations supported by observations, post-processed outputs, and photographic acquisition parameters." .

#################################################################
# Classes
#################################################################

:CulturalProperty a owl:Class ;
    rdfs:label "Cultural Property" ;
    rdfs:comment "A cultural heritage object, asset, or entity that is the subject of a survey or investigation activity." .

:FeatureOfInterest a owl:Class ;
    rdfs:label "Feature of Interest" ;
    rdfs:comment "A specific part, aspect, area, or feature of a cultural property that is the subject of a survey or investigation activity." .

:SurveyActivity a owl:Class ;
    rdfs:label "Survey Activity" ;
    rdfs:comment "An activity performed to survey, document, acquire, or investigate a cultural property or one of its features." .

:Method a owl:Class ;
    rdfs:label "Method" ;
    rdfs:comment "A method or investigation technique adopted in a survey or investigation activity." .

:Observation a owl:Class ;
    rdfs:label "Observation" ;
    rdfs:comment "An observation produced during a survey or investigation activity." .

:Expert a owl:Class ;
    rdfs:label "Expert" ;
    rdfs:comment "A person who participates in a survey or investigation activity." .

:Participation a owl:Class ;
    rdfs:label "Participation" ;
    rdfs:comment "A participation situation linking an expert to a survey or investigation activity and specifying the role held." .

:Role a owl:Class ;
    rdfs:label "Role" ;
    rdfs:comment "A role held by an expert in the context of participation in a survey or investigation activity." .

:Hardware a owl:Class ;
    rdfs:label "Hardware" ;
    rdfs:comment "A hardware device or instrument used to conduct a survey or investigation activity." .

:Software a owl:Class ;
    rdfs:label "Software" ;
    rdfs:comment "A software application or tool used to conduct a survey or investigation activity." .

:TechnicalConfiguration a owl:Class ;
    rdfs:label "Technical Configuration" ;
    rdfs:comment "A technical configuration used in conducting a survey or investigation activity." .

:Interpretation a owl:Class ;
    rdfs:label "Interpretation" ;
    rdfs:comment "An interpretation or claim supported by one or more observations." .

:PostProcessedOutput a owl:Class ;
    rdfs:label "Post-Processed Output" ;
    rdfs:comment "An output generated after processing data from an acquisition or observation activity, such as an interactive image, spectrum, or dataset." .

:PhotographicAcquisitionCycle a owl:Class ;
    rdfs:label "Photographic Acquisition Cycle" ;
    rdfs:comment "A photographic acquisition cycle carried out within a survey activity." ;
    rdfs:subClassOf :SurveyActivity .

:AcquisitionParameterSet a owl:Class ;
    rdfs:label "Acquisition Parameter Set" ;
    rdfs:comment "A set of acquisition parameters characterising a specific photographic acquisition cycle." .

#################################################################
# Object Properties
#################################################################

:performedOnCulturalProperty a owl:ObjectProperty ;
    rdfs:label "performed on cultural property" ;
    rdfs:comment "Relates a survey activity to the cultural property on which it was performed." ;
    rdfs:domain :SurveyActivity ;
    rdfs:range :CulturalProperty .

:performedOnFeature a owl:ObjectProperty ;
    rdfs:label "performed on feature" ;
    rdfs:comment "Relates a survey activity to the specific feature of interest on which it was performed." ;
    rdfs:domain :SurveyActivity ;
    rdfs:range :FeatureOfInterest .

:featureOfProperty a owl:ObjectProperty ;
    rdfs:label "feature of property" ;
    rdfs:comment "Relates a feature of interest to the cultural property of which it is a feature." ;
    rdfs:domain :FeatureOfInterest ;
    rdfs:range :CulturalProperty .

:adoptedMethod a owl:ObjectProperty ;
