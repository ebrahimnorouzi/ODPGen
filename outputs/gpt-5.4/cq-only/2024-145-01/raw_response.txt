```turtle
@prefix owl:  <http://www.w3.org/2002/07/owl#> .
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#> .
@prefix :     <http://example.org/odp#> .

:Ontology a owl:Ontology ;
    rdfs:label "Court and Check Payment ODP" ;
    rdfs:comment "An ontology design pattern for representing district courts and their states and districts, court authority relations, and bad-check-related check payment information needed to answer the stated competency questions." .

:State a owl:Class ;
    rdfs:label "State" ;
    rdfs:comment "A state, including a state of India, used to locate courts and contain districts." .

:District a owl:Class ;
    rdfs:label "District" ;
    rdfs:comment "An administrative district that is part of a state." .

:Court a owl:Class ;
    rdfs:label "Court" ;
    rdfs:comment "A court that may have authority over another court." .

:DistrictCourt a owl:Class ;
    rdfs:label "District Court" ;
    rdfs:comment "A court that is a district court." ;
    rdfs:subClassOf :Court .

:CheckTransactionType a owl:Class ;
    rdfs:label "Check Transaction Type" ;
    rdfs:comment "A type of transaction associated with a check or check payment." .

:TransactionStatus a owl:Class ;
    rdfs:label "Transaction Status" ;
    rdfs:comment "The status of a transaction or check payment." .

:CheckPayment a owl:Class ;
    rdfs:label "Check Payment" ;
    rdfs:comment "A payment made by check, with a transaction type, status, and payment time." .

:BadCheck a owl:Class ;
    rdfs:label "Bad Check" ;
    rdfs:comment "A check that is bad and can be associated with a check payment and a check transaction type." .

:locatedInState a owl:ObjectProperty ;
    rdfs:label "located in state" ;
    rdfs:comment "Relates a district court to the state in which it is located." ;
    rdfs:domain :DistrictCourt ;
    rdfs:range :State .

:hasDistrict a owl:ObjectProperty ;
    rdfs:label "has district" ;
    rdfs:comment "Relates a state to a district that is part of it." ;
    rdfs:domain :State ;
    rdfs:range :District .

:hasAuthorityOver a owl:ObjectProperty ;
    rdfs:label "has authority over" ;
    rdfs:comment "Relates a court to another court over which it has authority." ;
    rdfs:domain :Court ;
    rdfs:range :Court .

:hasCheckTransactionType a owl:ObjectProperty ;
    rdfs:label "has check transaction type" ;
    rdfs:comment "Relates a bad check or a check payment to its check transaction type." ;
    rdfs:range :CheckTransactionType .

:isPaymentOfBadCheck a owl:ObjectProperty ;
    rdfs:label "is payment of bad check" ;
    rdfs:comment "Relates a check payment to the bad check of which it is a payment." ;
    rdfs:domain :CheckPayment ;
    rdfs:range :BadCheck .

:hasTransactionStatus a owl:ObjectProperty ;
    rdfs:label "has transaction status" ;
    rdfs:comment "Relates a check payment to its transaction status." ;
    rdfs:domain :CheckPayment ;
    rdfs:range :TransactionStatus .

:paymentDateTime a owl:DatatypeProperty ;
    rdfs:label "payment date time" ;
    rdfs:comment "The date and time when a check payment occurred." ;
    rdfs:domain :CheckPayment ;
    rdfs:range xsd:dateTime .
```

| Competency Question | Classes | Properties |
|---|---|---|
| List all district courts in a given state. | `:DistrictCourt`, `:State` | `:locatedInState` |
| In which state is a given district court instance located? | `:DistrictCourt`, `:State` | `:locatedInState` |
| How many districts are in a given state of India? | `:State`, `:District` | `:hasDistrict` |
| Which court has authority over a given court? | `:Court` | `:hasAuthorityOver` |
