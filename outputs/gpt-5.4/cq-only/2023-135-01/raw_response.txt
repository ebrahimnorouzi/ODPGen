```turtle
@prefix owl:  <http://www.w3.org/2002/07/owl#> .
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#> .
@prefix :     <http://example.org/odp#> .

:Ontology a owl:Ontology ;
    rdfs:label "Pseudonym authorship and river-park crossing ODP" ;
    rdfs:comment "An ontology design pattern for representing authors using alternative identities such as pseudonyms when acting as writers of works, and for representing river trajectories that cross national parks." .

:Person a owl:Class ;
    rdfs:label "Person" ;
    rdfs:comment "A human individual." .

:AlternativeIdentity a owl:Class ;
    rdfs:label "Alternative identity" ;
    rdfs:comment "An alternative identity used by a person, such as a pseudonym." .

:Pseudonym a owl:Class ;
    rdfs:label "Pseudonym" ;
    rdfs:comment "An alternative identity used by a person for publication or authorship." ;
    rdfs:subClassOf :AlternativeIdentity .

:WriterRole a owl:Class ;
    rdfs:label "Writer role" ;
    rdfs:comment "The role of being a writer or author of a work." .

:Work a owl:Class ;
    rdfs:label "Work" ;
    rdfs:comment "A created intellectual or artistic work, such as a book or a collection of poems." .

:CollectionOfPoems a owl:Class ;
    rdfs:label "Collection of poems" ;
    rdfs:comment "A work consisting of a collection of poems." ;
    rdfs:subClassOf :Work .

:River a owl:Class ;
    rdfs:label "River" ;
    rdfs:comment "A natural flowing watercourse." .

:Trajectory a owl:Class ;
    rdfs:label "Trajectory" ;
    rdfs:comment "The spatial path or course followed by a river." .

:NationalPark a owl:Class ;
    rdfs:label "National park" ;
    rdfs:comment "A protected natural park designated at national level." .

:usesAlternativeIdentity a owl:ObjectProperty ;
    rdfs:label "uses alternative identity" ;
    rdfs:comment "Relates a person to an alternative identity that the person uses." ;
    rdfs:domain :Person ;
    rdfs:range :AlternativeIdentity .

:assumesRole a owl:ObjectProperty ;
    rdfs:label "assumes role" ;
    rdfs:comment "Relates a person to a role that the person assumes." ;
    rdfs:domain :Person ;
    rdfs:range :WriterRole .

:usesIdentityInRole a owl:ObjectProperty ;
    rdfs:label "uses identity in role" ;
    rdfs:comment "Relates a person to an alternative identity used when assuming a specific role such as writer." ;
    rdfs:domain :Person ;
    rdfs:range :AlternativeIdentity .

:publishesWork a owl:ObjectProperty ;
    rdfs:label "publishes work" ;
    rdfs:comment "Relates an alternative identity to a work published under that identity." ;
    rdfs:domain :AlternativeIdentity ;
    rdfs:range :Work .

:hasTitle a owl:DatatypeProperty ;
    rdfs:label "has title" ;
    rdfs:comment "Relates a work to its title." ;
    rdfs:domain :Work ;
    rdfs:range xsd:string .

:hasTrajectory a owl:ObjectProperty ;
    rdfs:label "has trajectory" ;
    rdfs:comment "Relates a river to its trajectory." ;
    rdfs:domain :River ;
    rdfs:range :Trajectory .

:crossesNationalPark a owl:ObjectProperty ;
    rdfs:label "crosses national park" ;
    rdfs:comment "Relates a trajectory to a national park that it crosses." ;
    rdfs:domain :Trajectory ;
    rdfs:range :NationalPark .
```
