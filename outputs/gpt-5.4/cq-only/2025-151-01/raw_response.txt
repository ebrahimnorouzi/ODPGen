```turtle
@prefix owl:  <http://www.w3.org/2002/07/owl#> .
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ndf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#> .
@prefix :     <http://example.org/odp#> .

:Ontology a owl:Ontology ;
    rdfs:label "Evaluation Request ODP" ;
    rdfs:comment "An ontology design pattern for representing an evaluation request, including the requested action, issuing party, target asset, issue time, and additional contextual information that constrains the request." .

:EvaluationRequest a owl:Class ;
    rdfs:label "Evaluation Request" ;
    rdfs:comment "A request that asks for an evaluation to be performed on a target asset." .

:RequestedAction a owl:Class ;
    rdfs:label "Requested Action" ;
    rdfs:comment "An action that is being requested by an evaluation request." .

:Party a owl:Class ;
    rdfs:label "Party" ;
    rdfs:comment "An agent or organization that can issue an evaluation request." .

:Asset a owl:Class ;
    rdfs:label "Asset" ;
    rdfs:comment "A target entity that is the subject of an evaluation request." .

:ContextualInformation a owl:Class ;
    rdfs:label "Contextual Information" ;
    rdfs:comment "Additional information included in an evaluation request to further constrain or qualify it." .

:hasRequestedAction a owl:ObjectProperty ;
    rdfs:label "has requested action" ;
    rdfs:comment "Relates an evaluation request to the action being requested." ;
    rdfs:domain :EvaluationRequest ;
    rdfs:range :RequestedAction .

:issuedBy a owl:ObjectProperty ;
    rdfs:label "issued by" ;
    rdfs:comment "Relates an evaluation request to the party that issued it." ;
    rdfs:domain :EvaluationRequest ;
    rdfs:range :Party .

:hasTargetAsset a owl:ObjectProperty ;
    rdfs:label "has target asset" ;
    rdfs:comment "Relates an evaluation request to the asset that is to be evaluated." ;
    rdfs:domain :EvaluationRequest ;
    rdfs:range :Asset .

:hasContextualInformation a owl:ObjectProperty ;
    rdfs:label "has contextual information" ;
    rdfs:comment "Relates an evaluation request to additional contextual information that further constrains or qualifies the request." ;
    rdfs:domain :EvaluationRequest ;
    rdfs:range :ContextualInformation .

:issuedAtTime a owl:DatatypeProperty ;
    rdfs:label "issued at time" ;
    rdfs:comment "Relates an evaluation request to the date and time at which it was issued." ;
    rdfs:domain :EvaluationRequest ;
    rdfs:range xsd:dateTime .
```
