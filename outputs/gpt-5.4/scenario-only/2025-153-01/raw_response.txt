```turtle
@prefix owl:  <http://www.w3.org/2002/07/owl#> .
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#> .
@prefix :     <http://example.org/odp#> .

:Ontology a owl:Ontology ;
    rdfs:label "Temporal Indirection via Persistent Identifier ODP" ;
    rdfs:comment "An ontology design pattern for modeling a stable persistent identifier that resolves over time to different time-specific referents, while current attributes are attached to the current referent and exposed through the stable identifier." .

:PersistentIdentifier a owl:Class ;
    rdfs:label "Persistent Identifier" ;
    rdfs:comment "A stable identifier that remains invariant over time while resolving to different referents at different times." .

:TimeSpecificReferent a owl:Class ;
    rdfs:label "Time-Specific Referent" ;
    rdfs:comment "An entity that is the concrete referent of a persistent identifier during a particular time period." .

:ResolutionPeriod a owl:Class ;
    rdfs:label "Resolution Period" ;
    rdfs:comment "A time interval during which a persistent identifier resolves to a particular time-specific referent." .

:MarketAttribute a owl:Class ;
    rdfs:label "Market Attribute" ;
    rdfs:comment "A real-time market attribute bound to the current contract instance and exposed through the persistent identifier." .

:LastTradePrice a owl:Class ;
    rdfs:subClassOf :MarketAttribute ;
    rdfs:label "Last Trade Price" ;
    rdfs:comment "A market attribute representing the most recent trade price of the current referent." .

:TradingVolume a owl:Class ;
    rdfs:subClassOf :MarketAttribute ;
    rdfs:label "Trading Volume" ;
    rdfs:comment "A market attribute representing the trading volume of the current referent." .

:SettlementValue a owl:Class ;
    rdfs:subClassOf :MarketAttribute ;
    rdfs:label "Settlement Value" ;
    rdfs:comment "A market attribute representing the settlement value of the current referent." .

:PersistentIdentifier owl:disjointWith :TimeSpecificReferent, :ResolutionPeriod, :MarketAttribute .
:TimeSpecificReferent owl:disjointWith :ResolutionPeriod, :MarketAttribute .
:ResolutionPeriod owl:disjointWith :MarketAttribute .

:hasResolutionPeriod a owl:ObjectProperty ;
    rdfs:domain :PersistentIdentifier ;
    rdfs:range :ResolutionPeriod ;
    rdfs:label "has resolution period" ;
    rdfs:comment "Relates a persistent identifier to a time interval during which it resolves to a particular referent." .

:forIdentifier a owl:ObjectProperty ;
    rdfs:domain :ResolutionPeriod ;
    rdfs:range :PersistentIdentifier ;
    rdfs:label "for identifier" ;
    rdfs:comment "Relates a resolution period to the persistent identifier whose resolution it specifies." ;
    owl:inverseOf :hasResolutionPeriod .

:resolvesToReferent a owl:ObjectProperty ;
    rdfs:domain :ResolutionPeriod ;
    rdfs:range :TimeSpecificReferent ;
    rdfs:label "resolves to referent" ;
    rdfs:comment "Relates a resolution period to the time-specific referent designated by the persistent identifier during that period." .

:isReferentInPeriod a owl:ObjectProperty ;
    rdfs:domain :TimeSpecificReferent ;
    rdfs:range :ResolutionPeriod ;
    rdfs:label "is referent in period" ;
    rdfs:comment "Relates a time-specific referent to a resolution period in which it is designated by a persistent identifier." ;
    owl:inverseOf :resolvesToReferent .

:hasCurrentAttribute a owl:ObjectProperty ;
    rdfs:domain :TimeSpecificReferent ;
    rdfs:range :MarketAttribute ;
    rdfs:label "has current attribute" ;
    rdfs:comment "Relates the current time-specific referent to a market attribute bound to it." .

:isAttributeOf a owl:ObjectProperty ;
    rdfs:domain :MarketAttribute ;
    rdfs:range :TimeSpecificReferent ;
    rdfs:label "is attribute of" ;
    rdfs:comment "Relates a market attribute to the time-specific referent to which it is bound." ;
    owl:inverseOf :hasCurrentAttribute .

    rdfs:domain :ResolutionPeriod ;
    rdfs:range xsd:dateTime ;
