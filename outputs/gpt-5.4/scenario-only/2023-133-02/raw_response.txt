```turtle
@prefix owl:  <http://www.w3.org/2002/07/owl#> .
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#> .
@prefix :     <http://example.org/odp#> .

:Ontology a owl:Ontology ;
    rdfs:label "kHealth-Asthma causal monitoring pattern" ;
    rdfs:comment "An ontology design pattern for representing pediatric asthma self-monitoring data about symptoms, medication intake, outdoor environmental features, and causal relations with effect weights among these factors." .

### Classes

:Patient a owl:Class ;
    rdfs:label "Patient" ;
    rdfs:comment "A pediatric asthma patient being continuously self-monitored in the kHealth-Asthma use case." .

:ObservationItem a owl:Class ;
    rdfs:label "Observation item" ;
    rdfs:comment "A monitored factor in the asthma use case, such as a symptom, a medication intake, or an outdoor environmental feature." .

:Symptom a owl:Class ;
    rdfs:subClassOf :ObservationItem ;
    rdfs:label "Symptom" ;
    rdfs:comment "A symptom experienced by a patient, such as cough, chest tightness, wheezing, or breathing pattern." .

:MedicationIntake a owl:Class ;
    rdfs:subClassOf :ObservationItem ;
    rdfs:label "Medication intake" ;
    rdfs:comment "An intake of medication by a patient as part of asthma self-monitoring." .

:OutdoorEnvironmentalFeature a owl:Class ;
    rdfs:subClassOf :ObservationItem ;
    rdfs:label "Outdoor environmental feature" ;
    rdfs:comment "An outdoor environmental factor relevant to asthma, such as pollen or air quality index." .

:CausalRelation a owl:Class ;
    rdfs:label "Causal relation" ;
    rdfs:comment "A relation representing that one monitored factor causally affects another, with an associated effect weight." .

### Disjointness

:Symptom owl:disjointWith :MedicationIntake, :OutdoorEnvironmentalFeature .

:MedicationIntake owl:disjointWith :OutdoorEnvironmentalFeature .

### Object Properties

:hasSymptom a owl:ObjectProperty ;
    rdfs:domain :Patient ;
    rdfs:range :Symptom ;
    rdfs:label "has symptom" ;
    rdfs:comment "Relates a patient to a symptom that is observed for that patient." .

:hasMedicationIntake a owl:ObjectProperty ;
    rdfs:domain :Patient ;
    rdfs:range :MedicationIntake ;
    rdfs:label "has medication intake" ;
    rdfs:comment "Relates a patient to a medication intake recorded for that patient." .

:hasOutdoorEnvironmentalFeature a owl:ObjectProperty ;
    rdfs:domain :Patient ;
    rdfs:range :OutdoorEnvironmentalFeature ;
    rdfs:label "has outdoor environmental feature" ;
    rdfs:comment "Relates a patient to an outdoor environmental feature considered in that patient's monitoring context." .

:hasCause a owl:ObjectProperty ;
    rdfs:domain :CausalRelation ;
    rdfs:range :ObservationItem ;
    rdfs:label "has cause" ;
    rdfs:comment "Relates a causal relation to the monitored factor that acts as the cause." .

:hasEffect a owl:ObjectProperty ;
    rdfs:domain :CausalRelation ;
    rdfs:range :ObservationItem ;
    rdfs:label "has effect" ;
    rdfs:comment "Relates a causal relation to the monitored factor that is affected." .

### Datatype Properties

:hasEffectWeight a owl:DatatypeProperty ;
    rdfs:domain :CausalRelation ;
    rdfs:range xsd:decimal ;
    rdfs:label "has effect weight" ;
    rdfs:comment "Associates a causal relation with a numeric weight expressing the strength of the causal effect." .
