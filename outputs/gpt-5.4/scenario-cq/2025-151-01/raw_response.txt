```turtle
@prefix owl:  <http://www.w3.org/2002/07/owl#> .
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#> .
@prefix :     <http://example.org/odp#> .

:Ontology a owl:Ontology ;
  rdfs:label "Evaluation Request Context ODP" ;
  rdfs:comment "An ontology design pattern for representing an evaluation request as a formal description of a requested action by an assignee on a target asset, together with temporal, spatial, role, and request-capability context used to constrain evaluation." .

:EvaluationRequest a owl:Class ;
  rdfs:label "Evaluation Request" ;
  rdfs:comment "A formal description of a requested action by an assignee on a target asset, optionally enriched with contextual information for evaluation." .

:Action a owl:Class ;
  rdfs:label "Action" ;
  rdfs:comment "An action requested in an evaluation request." .

:Party a owl:Class ;
  rdfs:label "Party" ;
  rdfs:comment "An entity that can issue an evaluation request." .

:Assignee a owl:Class ;
  rdfs:label "Assignee" ;
  rdfs:comment "The party on whose behalf or about whom the requested action is described in the evaluation request." ;
  rdfs:subClassOf :Party .

:Asset a owl:Class ;
  rdfs:label "Asset" ;
  rdfs:comment "A target asset to which the requested action applies." .

:ContextInformation a owl:Class ;
  rdfs:label "Context Information" ;
  rdfs:comment "Additional contextual information included in an evaluation request to further constrain evaluation." .

:TemporalContext a owl:Class ;
  rdfs:label "Temporal Context" ;
  rdfs:comment "Context information concerning time, such as weekday-related constraints." ;
  rdfs:subClassOf :ContextInformation .

:SpatialContext a owl:Class ;
  rdfs:label "Spatial Context" ;
  rdfs:comment "Context information concerning location, such as being from the EU." ;
  rdfs:subClassOf :ContextInformation .

:RoleContext a owl:Class ;
  rdfs:label "Role Context" ;
  rdfs:comment "Context information concerning role, such as employee status." ;
  rdfs:subClassOf :ContextInformation .

:RequestCapabilityContext a owl:Class ;
  rdfs:label "Request Capability Context" ;
  rdfs:comment "Context information concerning request capabilities relevant to evaluation." ;
  rdfs:subClassOf :ContextInformation .

:issuedAt a owl:DatatypeProperty ;
  rdfs:label "issued at" ;
  rdfs:comment "Relates an evaluation request to the date-time at which it was issued." ;
  rdfs:domain :EvaluationRequest ;
  rdfs:range xsd:dateTime .

:hasRequestedAction a owl:ObjectProperty ;
  rdfs:label "has requested action" ;
  rdfs:comment "Relates an evaluation request to the action being requested." ;
  rdfs:domain :EvaluationRequest ;
  rdfs:range :Action .

:issuedBy a owl:ObjectProperty ;
  rdfs:label "issued by" ;
  rdfs:comment "Relates an evaluation request to the party issuing it." ;
  rdfs:domain :EvaluationRequest ;
  rdfs:range :Party .

:hasAssignee a owl:ObjectProperty ;
  rdfs:label "has assignee" ;
  rdfs:comment "Relates an evaluation request to the assignee involved in the requested action." ;
  rdfs:domain :EvaluationRequest ;
  rdfs:range :Assignee .

:hasTargetAsset a owl:ObjectProperty ;
  rdfs:label "has target asset" ;
  rdfs:comment "Relates an evaluation request to the target asset of the requested action." ;
  rdfs:domain :EvaluationRequest ;
  rdfs:range :Asset .

:hasContextInformation a owl:ObjectProperty ;
  rdfs:label "has context information" ;
  rdfs:comment "Relates an evaluation request to additional contextual information that constrains evaluation." ;
  rdfs:domain :EvaluationRequest ;
  rdfs:range :ContextInformation .

:hasTemporalContext a owl:ObjectProperty ;
  rdfs:label "has temporal context" ;
  rdfs:comment "Relates an evaluation request to temporal context information." ;
  rdfs:subPropertyOf :hasContextInformation ;
  rdfs:domain :EvaluationRequest ;
  rdfs:range :TemporalContext .
