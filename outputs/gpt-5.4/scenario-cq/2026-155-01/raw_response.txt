:Ontology a owl:Ontology ;
  rdfs:label "Cultural Heritage Survey ODP" ;
  rdfs:comment "An ontology design pattern for representing scientific survey and investigation activities on cultural heritage entities, including observations, methods, participants and roles, equipment and technical configurations, interpretations, post-processed outputs, and photographic acquisition parameters." .

:SurveyActivity a owl:Class ;
  rdfs:label "Survey activity" ;
  rdfs:comment "A scientific survey or investigation activity performed on a cultural heritage entity or one of its features." .

:CulturalProperty a owl:Class ;
  rdfs:label "Cultural property" ;
  rdfs:comment "A cultural heritage entity that is the target of a survey or investigation activity." .

:FeatureOfInterest a owl:Class ;
  rdfs:label "Feature of interest" ;
  rdfs:comment "A specific feature, part, or aspect of a cultural property that is targeted by a survey or investigation activity." .

:Method a owl:Class ;
  rdfs:label "Method" ;
  rdfs:comment "A method or investigation technique adopted in a survey or investigation activity." .

:Observation a owl:Class ;
  rdfs:label "Observation" ;
  rdfs:comment "An observation produced during a survey or investigation activity." .

:Expert a owl:Class ;
  rdfs:label "Expert" ;
  rdfs:comment "A person or specialist participating in a survey or investigation activity." .

:Participation a owl:Class ;
  rdfs:label "Participation" ;
  rdfs:comment "A participation situation linking an expert to a survey activity and specifying the role held in that activity." .

:Role a owl:Class ;
  rdfs:label "Role" ;
  rdfs:comment "A role played by an expert in a survey or investigation activity." .

:EquipmentConfiguration a owl:Class ;
  rdfs:label "Equipment configuration" ;
  rdfs:comment "A technical configuration used to conduct a survey or investigation activity, including hardware, software, and related settings." .

:Hardware a owl:Class ;
  rdfs:label "Hardware" ;
  rdfs:comment "A hardware component used in an equipment configuration for a survey or investigation activity." .

:Software a owl:Class ;
  rdfs:label "Software" ;
  rdfs:comment "A software component used in an equipment configuration for a survey or investigation activity." .

:TechnicalConfiguration a owl:Class ;
  rdfs:label "Technical configuration" ;
  rdfs:comment "A technical setup or configuration element used to conduct a survey or investigation activity." .

:Interpretation a owl:Class ;
  rdfs:label "Interpretation" ;
  rdfs:comment "An interpretation or claim supported by observations produced during a survey." .

:PostProcessedOutput a owl:Class ;
  rdfs:label "Post-processed output" ;
  rdfs:comment "An output produced by post-processing, such as an interactive image, spectrum, or dataset, and related to underlying acquisition or observation activities." .

:PhotographicAcquisitionCycle a owl:Class ;
  rdfs:label "Photographic acquisition cycle" ;
  rdfs:comment "A photographic acquisition cycle occurring within a survey and characterised by acquisition parameters." .

:AcquisitionParameterSet a owl:Class ;
  rdfs:label "Acquisition parameter set" ;
  rdfs:comment "A set of acquisition parameters characterising a specific photographic acquisition cycle." .

:performedOnProperty a owl:ObjectProperty ;
  rdfs:label "performed on property" ;
  rdfs:comment "Relates a survey activity to the cultural property on which it was performed." ;
  rdfs:domain :SurveyActivity ;
  rdfs:range :CulturalProperty .

:performedOnFeature a owl:ObjectProperty ;
  rdfs:label "performed on feature" ;
  rdfs:comment "Relates a survey activity to the specific feature of interest on which it was performed." ;
  rdfs:domain :SurveyActivity ;
  rdfs:range :FeatureOfInterest .

:featureOfProperty a owl:ObjectProperty ;
  rdfs:label "feature of property" ;
  rdfs:comment "Relates a feature of interest to the cultural property of which it is a feature." ;
