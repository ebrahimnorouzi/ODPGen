```turtle
@prefix owl:  <http://www.w3.org/2002/07/owl#> .
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ndf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#> .
@prefix :     <http://example.org/odp#> .

:Ontology a owl:Ontology ;
  rdfs:label "Cultural Heritage Survey ODP" ;
  rdfs:comment "An ontology design pattern for representing scientific survey activities on cultural heritage entities, including surveys, observations, methods, expert participation, equipment and technical configurations, interpretations, post-processing outputs, and photographic acquisition parameters." .

#################################################################
# Classes
#################################################################

:SurveyActivity a owl:Class ;
  rdfs:label "Survey activity" ;
  rdfs:comment "A scientific survey or investigation activity performed on a cultural heritage entity or one of its features." .

:CulturalProperty a owl:Class ;
  rdfs:label "Cultural property" ;
  rdfs:comment "A cultural heritage entity that is the target of a survey or investigation activity." .

:FeatureOfInterest a owl:Class ;
  rdfs:label "Feature of interest" ;
  rdfs:comment "A specific feature or part of a cultural property that is targeted by a survey or investigation activity." .

:Method a owl:Class ;
  rdfs:label "Method" ;
  rdfs:comment "A method or investigation technique adopted in a survey or investigation activity." .

:Observation a owl:Class ;
  rdfs:label "Observation" ;
  rdfs:comment "An observation produced during a survey or investigation activity." .

:Expert a owl:Class ;
  rdfs:label "Expert" ;
  rdfs:comment "A person participating in a survey or investigation activity." .

:Participation a owl:Class ;
  rdfs:label "Participation" ;
  rdfs:comment "A participation situation linking an expert to a survey activity and specifying the role held." .

:Role a owl:Class ;
  rdfs:label "Role" ;
  rdfs:comment "A role held by an expert in the context of participation in a survey activity." .

:EquipmentConfiguration a owl:Class ;
  rdfs:label "Equipment configuration" ;
  rdfs:comment "A technical configuration used to conduct a survey or investigation activity." .

:Hardware a owl:Class ;
  rdfs:label "Hardware" ;
  rdfs:comment "A hardware component used within an equipment configuration for a survey activity." .

:Software a owl:Class ;
  rdfs:label "Software" ;
  rdfs:comment "A software component used within an equipment configuration for a survey activity." .

:TechnicalConfiguration a owl:Class ;
  rdfs:label "Technical configuration" ;
  rdfs:comment "A technical setup or configuration detail used within an equipment configuration for a survey activity." .

:Interpretation a owl:Class ;
  rdfs:label "Interpretation" ;
  rdfs:comment "An interpretation or claim supported by observations produced during a survey." .

:PostProcessedOutput a owl:Class ;
  rdfs:label "Post-processed output" ;
  rdfs:comment "An output produced by post-processing and related to an underlying acquisition or observation activity." .

:AcquisitionActivity a owl:Class ;
  rdfs:label "Acquisition activity" ;
  rdfs:comment "An acquisition activity carried out as part of a survey and underlying observations or post-processed outputs." ;
  rdfs:subClassOf :SurveyActivity .

:PhotographicAcquisitionCycle a owl:Class ;
  rdfs:label "Photographic acquisition cycle" ;
  rdfs:comment "A photographic acquisition cycle within a survey, characterised by acquisition parameters." ;
  rdfs:subClassOf :AcquisitionActivity .

:AcquisitionParameterSet a owl:Class ;
  rdfs:label "Acquisition parameter set" ;
  rdfs:comment "A set of acquisition parameters characterising a specific photographic acquisition cycle." .

#################################################################
# Object Properties
#################################################################

:performedOnProperty a owl:ObjectProperty ;
  rdfs:label "performed on property" ;
  rdfs:comment "Relates a survey activity to the cultural property on which it was performed." ;
  rdfs:domain :SurveyActivity ;
  rdfs:range :CulturalProperty .

:performedOnFeature a owl:ObjectProperty ;
  rdfs:label "performed on feature" ;
  rdfs:comment "Relates a survey activity to the specific feature of interest on which it was performed." ;
  rdfs:domain :SurveyActivity ;
  rdfs:range :FeatureOfInterest .
