```turtle
@prefix owl:  <http://www.w3.org/2002/07/owl#> .
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#> .
@prefix :     <http://example.org/odp#> .

:Ontology a owl:Ontology ;
    rdfs:label "Court hierarchy and bad check regulation ODP" ;
    rdfs:comment "An ontology design pattern for representing court hierarchies by state and district, and bad check related check payment information including transaction type, payment time, and transaction status." .

:State a owl:Class ;
    rdfs:label "State" ;
    rdfs:comment "A state administrative unit used to locate courts and contain districts." .

:District a owl:Class ;
    rdfs:label "District" ;
    rdfs:comment "A district administrative unit that is part of a state." .

:Court a owl:Class ;
    rdfs:label "Court" ;
    rdfs:comment "A court within a legal hierarchy." .

:DistrictCourt a owl:Class ;
    rdfs:label "District court" ;
    rdfs:comment "A court at the district level." ;
    rdfs:subClassOf :Court .

:CheckTransaction a owl:Class ;
    rdfs:label "Check transaction" ;
    rdfs:comment "A transaction involving a check." .

:BadCheck a owl:Class ;
    rdfs:label "Bad check" ;
    rdfs:comment "A check transaction that is classified as a bad check." ;
    rdfs:subClassOf :CheckTransaction .

:CheckTransactionType a owl:Class ;
    rdfs:label "Check transaction type" ;
    rdfs:comment "A type assigned to a check transaction." .

:CheckPayment a owl:Class ;
    rdfs:label "Check payment" ;
    rdfs:comment "A payment event or record associated with a check transaction." .

:TransactionStatus a owl:Class ;
    rdfs:label "Transaction status" ;
    rdfs:comment "The status of a check payment transaction." .

:locatedInState a owl:ObjectProperty ;
    rdfs:label "located in state" ;
    rdfs:comment "Relates a court to the state in which it is located." ;
    rdfs:domain :Court ;
    rdfs:range :State .

:hasDistrict a owl:ObjectProperty ;
    rdfs:label "has district" ;
    rdfs:comment "Relates a state to a district that is part of it." ;
    rdfs:domain :State ;
    rdfs:range :District .

:isDistrictOf a owl:ObjectProperty ;
    rdfs:label "is district of" ;
    rdfs:comment "Relates a district to the state of which it is part." ;
    rdfs:domain :District ;
    rdfs:range :State ;
    owl:inverseOf :hasDistrict .

:hasAuthorityOver a owl:ObjectProperty ;
    rdfs:label "has authority over" ;
    rdfs:comment "Relates a court to another court over which it has authority." ;
    rdfs:domain :Court ;
    rdfs:range :Court .

:hasTransactionType a owl:ObjectProperty ;
    rdfs:label "has transaction type" ;
    rdfs:comment "Relates a check transaction to its transaction type." ;
    rdfs:domain :CheckTransaction ;
    rdfs:range :CheckTransactionType .

:hasCheckPayment a owl:ObjectProperty ;
    rdfs:label "has check payment" ;
    rdfs:comment "Relates a check transaction, including a bad check, to an associated check payment." ;
    rdfs:domain :CheckTransaction ;
    rdfs:range :CheckPayment .

:hasTransactionStatus a owl:ObjectProperty ;
    rdfs:label "has transaction status" ;
    rdfs:comment "Relates a check payment to its transaction status." ;
    rdfs:domain :CheckPayment ;
    rdfs:range :TransactionStatus .

:paymentDateTime a owl:DatatypeProperty ;
    rdfs:label "payment date time" ;
    rdfs:comment "The date and time when a check payment occurred." ;
    rdfs:domain :CheckPayment ;
    rdfs:range xsd:dateTime .
```
