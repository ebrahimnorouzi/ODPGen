:Ontology a owl:Ontology ;

### Classes

:Policy a owl:Class ;
  rdfs:label "Policy" ;
  rdfs:comment "A policy being evaluated." .

:Rule a owl:Class ;
  rdfs:label "Rule" ;
  rdfs:comment "A rule that is part of a policy and whose exercise can be tracked during evaluation." .

:Party a owl:Class ;
  rdfs:label "Party" ;
  rdfs:comment "An agent or actor relevant to policy evaluation, such as a requester or other involved party." .

:PartyCollection a owl:Class ;
  rdfs:label "Party collection" ;
  rdfs:comment "A collection of parties associated with a policy." .

:Asset a owl:Class ;
  rdfs:label "Asset" ;
  rdfs:comment "An asset relevant to a policy, such as a video stream." .

:AssetCollection a owl:Class ;
  rdfs:label "Asset collection" ;
  rdfs:comment "A collection of assets associated with a policy." .

:Action a owl:Class ;
  rdfs:label "Action" ;
  rdfs:comment "An action that may be performed or attempted in relation to a rule or policy." .

:Event a owl:Class ;
  rdfs:label "Event" ;
  rdfs:comment "An event that has occurred and is relevant to policy evaluation." .

:Recipient a owl:Class ;
  rdfs:label "Recipient" ;
  rdfs:comment "A recipient that can receive the result of an exercised rule." .

:RecipientCategory a owl:Class ;
  rdfs:label "Recipient category" ;
  rdfs:comment "A category of recipients that can receive the result of an exercised rule." .

:Payment a owl:Class ;
  rdfs:label "Payment" ;
  rdfs:comment "A financial payment relevant to policy evaluation." .

:Purpose a owl:Class ;
  rdfs:label "Purpose" ;
  rdfs:comment "A purpose for which access or use is requested." .

:TimeInterval a owl:Class ;
  rdfs:label "Time interval" ;
  rdfs:comment "A temporal interval, such as a segment of a media asset." .

:Location a owl:Class ;
  rdfs:label "Location" ;
  rdfs:comment "A location associated with a party." .
