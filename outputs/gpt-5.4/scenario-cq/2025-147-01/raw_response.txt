### Core ship concepts

:Ship a owl:Class ;
  rdfs:label "Ship" ;
  rdfs:comment "A vessel represented in the maritime domain." .

:ShipType a owl:Class ;
  rdfs:label "Ship type" ;
  rdfs:comment "A category or type of ship, such as container ship." .

:ContainerShip a owl:Class ;
  rdfs:subClassOf :Ship ;
  rdfs:label "Container ship" ;
  rdfs:comment "A ship designed or classified for carrying containers." .

:Identifier a owl:Class ;
  rdfs:label "Identifier" ;
  rdfs:comment "An identification entity used to clearly identify a ship." .

:DimensionSpecification a owl:Class ;
  rdfs:label "Dimension specification" ;
  rdfs:comment "A specification describing one or more physical dimensions of a ship." .

:Measurement a owl:Class ;
  rdfs:label "Measurement" ;
  rdfs:comment "A quantitative value together with its unit." .

:Unit a owl:Class ;
  rdfs:label "Unit" ;
  rdfs:comment "A unit used for a measurement." .

:Location a owl:Class ;
  rdfs:label "Location" ;
  rdfs:comment "A place or area associated with a ship, such as a typical location." .

:Organization a owl:Class ;
  rdfs:label "Organization" ;
  rdfs:comment "An organization that can own ships or act as a registry-related entity." .

:Person a owl:Class ;
  rdfs:label "Person" ;
  rdfs:comment "A person who can own a ship." .

:Owner a owl:Class ;
  owl:unionOf ( :Organization :Person ) ;
  rdfs:label "Owner" ;
  rdfs:comment "An owner of a ship, either an organization or a person." .

:FlagState a owl:Class ;
  rdfs:label "Flag state" ;
  rdfs:comment "A state or flag under which a ship is registered." .

  rdfs:domain :Identifier ;
