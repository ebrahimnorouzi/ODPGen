**1. Turtle Ontology**

```turtle
:Ontology a owl:Ontology ;
    rdfs:label "Registry Ontology" ;
    rdfs:comment "An ontology for modeling a data registry.".

:Registry a owl:Class ;
    rdfs:label "Registry" ;
    rdfs:comment "A collection of data resources.".

:Registry rdf:type owl:NamedIndividual ;
    :hasMaintainer?maintainer ;
    :creationDate?creationDate ;
    :dataType?dataType ;
    :storageLocation?storageLocation ;
    :dataPolicy?dataPolicy.

:Maintainer a owl:Class ;
    rdfs:label "Maintainer" ;
    rdfs:comment "The entity responsible for maintaining the registry.".

:Maintainer rdf:type owl:ObjectProperty ;
    rdfs:domain :Registry ;
    rdfs:range rdf:Resource ;
    rdfs:label "hasMaintainer" ;
    rdfs:comment "The maintainer of the registry.".

:CreationDate a owl:DatatypeProperty ;
    rdfs:domain :Registry ;
    rdfs:range xsd:dateTime ;
    rdfs:label "creationDate" ;
    rdfs:comment "The date when the registry was created or updated.".

:DataType a owl:Class ;
    rdfs:label "Data Type" ;
    rdfs:comment "The type of data available in the registry.".

:DataType rdf:type owl:Enum ;
    owl:someValuesFrom owl:Thing ;
    :Data1 rdf:PlainLiteral "Data Type 1" ;
    :Data2 rdf:PlainLiteral "Data Type 2" ;
    :Data3 rdf:PlainLiteral "Data Type 3".

:StorageLocation a owl:DatatypeProperty ;
    rdfs:domain :Registry ;
    rdfs:range xsd:string ;
    rdfs:label "storageLocation" ;
    rdfs:comment "The location where a specific type of data is stored.".

:DataPolicy a owl:Class ;
    rdfs:label "Data Policy" ;
    rdfs:comment "The policy associated with the data in the registry.".

:DataPolicy rdf:type owl:ObjectProperty ;
    rdfs:domain :Registry ;
    rdfs:range rdf:Resource ;
    rdfs:label "dataPolicy" ;
    rdfs:comment "The policy associated with the data in the registry.".
```
