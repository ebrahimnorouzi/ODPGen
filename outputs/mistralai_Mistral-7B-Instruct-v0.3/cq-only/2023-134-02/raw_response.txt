@prefix owl: <http://www.w3.org/2002/07/owl#>.
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>.

@prefix xsd: <http://www.w3.org/2001/XMLSchema#>.
@prefix : <http://example.org/odp#>.

:Ontology a owl:Ontology ;
  rdfs:label "Data Log Ontology" ;
  rdfs:comment "An ontology for modeling data logs with competency questions about actions, entities, timestamps, locations, and applications.".

# Classes
:Action a owl:Class ;
  rdfs:label "Action" ;
  rdfs:comment "The type of action being performed on the data.".

:Entity a owl:Class ;
  rdfs:label "Entity" ;
  rdfs:comment "The entity interacting with the data or publishing the log.".

:Publisher a owl:Class ;
  rdfs:subClassOf [ a owl:Restriction ;
                    owl:onProperty :publisher ;
                    owl:hasValue :Entity. ] ;
  rdfs:label "Publisher" ;
  rdfs:comment "The entity publishing the log.".

:Timestamp a owl:Class ;
  rdfs:label "Timestamp" ;
  rdfs:comment "The time when the log was issued.".

:Location a owl:Class ;
  rdfs:label "Location" ;
  rdfs:comment "The place where the data is being stored.".

:Application a owl:Class ;
  rdfs:label "Application" ;
  rdfs:comment "The application or service being used to generate the data.".

# Object Properties
:performedBy a owl:ObjectProperty ;
  rdfs:domain :Action ;
  rdfs:range :Entity ;
  rdfs:label "performedBy" ;
  rdfs:comment "The entity that performed the action.".

:publishedBy a owl:ObjectProperty ;
  rdfs:domain :Log ;
  rdfs:range :Publisher ;
  rdfs:label "publishedBy" ;
  rdfs:comment "The entity that published the log.".

:hasTimestamp a owl:ObjectProperty ;
  rdfs:domain :Log ;
  rdfs:range :Timestamp ;
  rdfs:label "hasTimestamp" ;
  rdfs:comment "The timestamp of the log.".

:isStoredAt a owl:ObjectProperty ;
  rdfs:domain :Location ;
  rdfs:range rdf:Resource ;
  rdfs:label "isStoredAt" ;
  rdfs:comment "The location where the data is being stored.".

:generatedBy a owl:ObjectProperty ;
  rdfs:domain :Data ;
  rdfs:range :Application ;
  rdfs:label "generatedBy" ;
  rdfs:comment "The application or service that generated the data.".

# Data Types
:Timestamp rdfs:subClassOf [ a xsd:dateTime ].

# Axioms
:Log a owl:Class ;
  rdfs:subClassOf [ a owl:Restriction ;
                    owl:onProperty :hasTimestamp ;
                    owl:hasValue ""^^xsd:dateTime. ] ;
  rdfs:label "Log" ;
  rdfs:comment "A log entry containing information about an action, the entity, the timestamp, the location, and the application.".
