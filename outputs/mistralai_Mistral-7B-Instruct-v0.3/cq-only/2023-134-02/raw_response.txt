```turtle
@prefix owl: <http://www.w3.org/2002/07/owl#>.
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>.
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#>.
@prefix xsd: <http://www.w3.org/2001/XMLSchema#>.
@prefix : <http://example.org/odp#>.

:Ontology a owl:Ontology ;
  rdfs:label "Data Interaction Log Ontology" ;
  rdfs:comment "An ontology for modeling data interaction logs.".

# Action
:Action rdf:Type owl:Class ;
  rdfs:label "Action" ;
  rdfs:comment "The type of action performed on the data.".

:Create a :Action ;
  rdfs:label "Create" ;
  rdfs:comment "The action of creating data.".

:Update a :Action ;
  rdfs:label "Update" ;
  rdfs:comment "The action of updating data.".

:Erase a :Action ;
  rdfs:label "Erase" ;
  rdfs:comment "The action of erasing data.".

# Interactor
:Interactor rdf:Type owl:Class ;
  rdfs:label "Interactor" ;
  rdfs:comment "The entity interacting with the data.".

# Publisher
:Publisher rdf:Type owl:Class ;
  rdfs:label "Publisher" ;
  rdfs:comment "The entity publishing the log.".

# Timestamp
:Timestamp rdf:Type owl:Class ;
  rdfs:label "Timestamp" ;
  rdfs:comment "The time when the log was issued.".

:Year xsd:dateTime "0000-01-01T00:00:00"^^xsd:dateTime ;
  rdfs:label "Year" ;
  rdfs:comment "The year part of the timestamp.".

:Month xsd:gYearMonth "0001-01"^^xsd:gYearMonth ;
  rdfs:label "Month" ;
  rdfs:comment "The month part of the timestamp.".

:Day xsd:gDay "01"^^xsd:gDay ;
  rdfs:label "Day" ;
  rdfs:comment "The day part of the timestamp.".

:Hour xsd:gYearMonthDayTime "00:00:00"^^xsd:gYearMonthDayTime ;
  rdfs:label "Hour" ;
  rdfs:comment "The hour part of the timestamp.".

:Minute xsd:gYearMonthDayTime "00:00"^^xsd:gYearMonthDayTime ;
  rdfs:label "Minute" ;
  rdfs:comment "The minute part of the timestamp.".

:Second xsd:gYearMonthDayTime "00"^^xsd:gYearMonthDayTime ;
  rdfs:label "Second" ;
  rdfs:comment "The second part of the timestamp.".

# Storage
:Storage rdf:Type owl:Class ;
  rdfs:label "Storage" ;
  rdfs:comment "Where the data is being stored.".

# Application/Service
:Application rdf:Type owl:Class ;
  rdfs:label "Application/Service" ;
  rdfs:comment "The application or service generating the data.".

# Object Properties
:performs :Action rdf:Type owl:ObjectProperty ;
  rdfs:domain :Interactor ;
  rdfs:range :Action ;
  rdfs:label "performs" ;
  rdfs:comment "The action performed by the interactor.".

:publishes :Log rdf:Type owl:ObjectProperty ;
