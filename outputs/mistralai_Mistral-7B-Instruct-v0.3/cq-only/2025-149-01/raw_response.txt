# Database
:Database a owl:Class ;
  rdfs:label "Database" ;
  rdfs:comment "A first-class schema entity representing a relational database.".

:hasTable rdf:Property ;
  rdfs:domain :Database ;
  rdfs:range :Table ;
  rdfs:label "hasTable" ;
  rdfs:comment "Relates a database to its tables.".

  rdfs:label "Table" ;

:hasColumn rdf:Property ;
  rdfs:domain :Table ;
  rdfs:range :Column ;
  rdfs:label "hasColumn" ;

:hasPrimaryKey rdf:Property ;
  rdfs:domain :Table ;
  rdfs:range :Column ;
  rdfs:label "hasPrimaryKey" ;
  rdfs:comment "Relates a table to its primary key column(s).".

:hasForeignKey rdf:Property ;
  rdfs:domain :Table ;
  rdfs:range :ForeignKey ;
  rdfs:label "hasForeignKey" ;
  rdfs:comment "Relates a table to its foreign key(s).".

:ForeignKey a owl:Class ;
