```turtle
@prefix owl: <http://www.w3.org/2002/07/owl#>.
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>.
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#>.
@prefix xsd: <http://www.w3.org/2001/XMLSchema#>.
@prefix : <http://example.org/odp#>.

:Ontology a owl:Ontology ;
  rdfs:label "Ontology Design Pattern" ;
  rdfs:comment "An ontology for answering competency questions about courts and checks.".

# Court ontology
:Court a owl:Class ;
  rdfs:label "Court" ;
  rdfs:comment "A court of law.".

:DistrictCourt a owl:Class ;
  rdfs:subClassOf [ a owl:Restriction ;
                    owl:onProperty :hasCourtType ;
                    owl:allValuesFrom (:DistrictCourtType "District Court") ] ;
  rdfs:label "District Court" ;
  rdfs:comment "A district court of law.".

:StateCourt a owl:Class ;
  rdfs:subClassOf [ a owl:Restriction ;
                    owl:onProperty :hasCourtType ;
                    owl:allValuesFrom (:StateCourtType "State Court") ] ;
  rdfs:label "State Court" ;
  rdfs:comment "A state court of law.".

:CourtType a owl:Class ;
  rdfs:label "Court Type" ;
  rdfs:comment "The type of a court.".

:hasCourtType a owl:ObjectProperty ;
  rdfs:domain :Court ;
  rdfs:range :CourtType ;
  rdfs:label "Has Court Type" ;
  rdfs:comment "The type of a court.".

:hasLocation a owl:ObjectProperty ;
  rdfs:domain :Court ;
  rdfs:range :Location ;
  rdfs:label "Has Location" ;
  rdfs:comment "The location of a court.".

:Location a owl:Class ;
  rdfs:label "Location" ;
  rdfs:comment "A geographical location.".

:State a owl:Class ;
  rdfs:label "State" ;
  rdfs:comment "A geographical state.".

:India a :State ;
  rdfs:label "India".

:Check a owl:Class ;

  rdfs:comment "A financial check.".

                    owl:onProperty :hasCheckType ;

  rdfs:label "Bad Check" ;
  rdfs:comment "A bad financial check.".

:CheckType a owl:Class ;
  rdfs:label "Check Type" ;
  rdfs:comment "The type of a check.".

:hasCheckType a owl:ObjectProperty ;
  rdfs:domain :Check ;
  rdfs:range :CheckType ;
  rdfs:label "Has Check Type" ;
  rdfs:comment "The type of a check.".

  rdfs:label "Has Transaction Type" ;
