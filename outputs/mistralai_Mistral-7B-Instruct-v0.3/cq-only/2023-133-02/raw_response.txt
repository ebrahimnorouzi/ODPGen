@prefix owl: <http://www.w3.org/2002/07/owl#>.
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>.
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#>.
@prefix xsd: <http://www.w3.org/2001/XMLSchema#>.
@prefix : <http://example.org/odp#>.

:Ontology a owl:Ontology ;
  rdfs:label "Environmental Allergy and Medication Intake Ontology" ;
  rdfs:comment "An ontology to model the relationship between environmental allergens, asthma symptoms, medication intake, and their effects.".

# Classes
:Patient a owl:Class ;
  rdfs:label "Patient" ;
  rdfs:comment "A person who may have environmental allergies and take medication.".

:EnvironmentalAllergen a owl:Class ;
  rdfs:label "Environmental Allergen" ;
  rdfs:comment "An allergen present in the outdoor environment, such as pollen or AQI.".

:AsthmaSymptom a owl:Class ;
  rdfs:label "Asthma Symptom" ;
  rdfs:comment "A symptom experienced by a patient with asthma.".

:Medication a owl:Class ;
  rdfs:label "Medication" ;
  rdfs:comment "A medication taken by a patient to alleviate asthma symptoms.".

:Cause a owl:Class ;
  rdfs:label "Cause" ;
  rdfs:comment "The reason for a certain event or state.".

:Effect a owl:Class ;
  rdfs:label "Effect" ;
  rdfs:comment "The result or consequence of a certain event or state.".

# Object Properties
:hasPatient a owl:ObjectProperty ;
  rdfs:domain :Medication ;
  rdfs:range :Patient ;
  rdfs:label "Has Patient" ;
  rdfs:comment "Relates a medication to the patient who takes it.".

:hasAllergen a owl:ObjectProperty ;
  rdfs:domain :EnvironmentalAllergen ;
  rdfs:range :Patient ;
  rdfs:label "Has Allergen" ;
  rdfs:comment "Relates a patient to the environmental allergen they are exposed to.".

:hasSymptom a owl:ObjectProperty ;
  rdfs:domain :Patient ;
  rdfs:range :AsthmaSymptom ;
  rdfs:label "Has Symptom" ;
  rdfs:comment "Relates a patient to the asthma symptom they experience.".

:causes a owl:ObjectProperty ;
  rdfs:domain :Cause ;
  rdfs:range :Effect ;
  rdfs:label "Causes" ;
  rdfs:comment "Relates a cause to its effect.".

# Datatype Properties
:intakeFrequency a owl:DatatypeProperty ;
  rdfs:domain :Medication ;
  rdfs:range xsd:nonNegativeInteger ;
  rdfs:label "Intake Frequency" ;
  rdfs:comment "The frequency at which a medication is taken.".

:severity a owl:DatatypeProperty ;
  rdfs:domain :AsthmaSymptom ;
  rdfs:range xsd:nonNegativeInteger ;
  rdfs:label "Severity" ;
  rdfs:comment "The severity of an asthma symptom.".
