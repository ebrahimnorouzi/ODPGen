1. Turtle ontology:

```turtle
:Ontology a owl:Ontology ;
    rdfs:label "kHealth-Asthma Ontology" ;
    rdfs:comment "An ontology for the kHealth-Asthma project, modeling asthma patients, symptoms, medications, and environmental factors.".

:Patient a owl:Class ;
    rdfs:label "Patient" ;
    rdfs:comment "A person suffering from asthma.".

:Symptom a owl:Class ;
    rdfs:label "Symptom" ;
    rdfs:comment "A health condition experienced by a patient.".

:Medication a owl:Class ;
    rdfs:label "Medication" ;
    rdfs:comment "A drug used to treat asthma symptoms.".

:Environment a owl:Class ;
    rdfs:label "Environment" ;
    rdfs:comment "The outdoor environment where the patient is located.".

:Allergen a owl:Class ;
    rdfs:label "Allergen" ;
    rdfs:comment "An outdoor environmental factor that triggers asthma symptoms.".

:Pollen rdf:type :Allergen ;
    rdfs:label "Pollen" ;
    rdfs:comment "Outdoor environmental allergen, such as pollen.".

:AQI rdf:type :Allergen ;
    rdfs:label "Air Quality Index" ;
    rdfs:comment "Outdoor environmental allergen, such as air quality index.".

:hasSymptom a owl:ObjectProperty ;
    rdfs:domain :Patient ;
    rdfs:range :Symptom ;
    rdfs:label "Has Symptom" ;
    rdfs:comment "Relates a patient to a symptom they are experiencing.".

:takesMedication a owl:ObjectProperty ;
    rdfs:domain :Patient ;
    rdfs:range :Medication ;
    rdfs:label "Takes Medication" ;
    rdfs:comment "Relates a patient to the medication they are taking.".

:isInEnvironment a owl:ObjectProperty ;
    rdfs:domain :Patient ;
    rdfs:range :Environment ;
    rdfs:label "Is in Environment" ;
    rdfs:comment "Relates a patient to the environment they are in.".

:hasAllergen a owl:ObjectProperty ;
    rdfs:domain :Environment ;
    rdfs:range :Allergen ;
    rdfs:label "Has Allergen" ;
    rdfs:comment "Relates an environment to an allergen present in it.".

:causesSymptom a owl:ObjectProperty ;
    rdfs:domain :Allergen ;
    rdfs:range :Symptom ;
    rdfs:label "Causes Symptom" ;
    rdfs:comment "Relates an allergen to a symptom it causes.".

:causesMedication a owl:ObjectProperty ;
    rdfs:domain :Symptom ;
    rdfs:range :Medication ;
    rdfs:label "Causes Medication" ;
    rdfs:comment "Relates a symptom to a medication it causes the patient to take.".

:effectWeight a xsd:decimal ;
    rdfs:label "Effect Weight" ;
    rdfs:comment "The weight of the effect of an allergen on a symptom or medication.".
```
