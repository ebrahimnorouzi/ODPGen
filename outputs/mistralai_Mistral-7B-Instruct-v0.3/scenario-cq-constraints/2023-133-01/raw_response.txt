    rdfs:comment "Event representing the rainy season.".

    rdfs:comment "Event representing the pavement is slippery.".

:CausalRelation a owl:ObjectProperty ;
    rdfs:domain :Event ;
    rdfs:range :Event ;
    rdfs:label "Causal Relation" ;
    rdfs:comment "Property for causal relations between events.".

:RainCausesPavementWet a :CausalRelation ;
