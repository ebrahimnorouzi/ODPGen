Here is the OWL ontology design pattern in Turtle syntax:

```turtle
@prefix owl: <http://www.w3.org/2002/07/owl#>.
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#>.
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#>.
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#>.
@prefix :     <http://example.org/odp#>.

:Ontology a owl:Ontology ;
    rdfs:label "Author Pseudonyms and Book Publication" ;
    rdfs:comment "An ontology for modeling authors adopting different names to publish books and their associated publications.".

:Author a owl:Class ;
    rdfs:label "Author" ;
    rdfs:comment "A person who writes books under different names.".

:Pseudonym a owl:Class ;
    rdfs:label "Pseudonym" ;
    rdfs:comment "An alternative name used by an author to publish books.".

:hasPseudonym rdf:property ;
    rdfs:domain :Author ;
    rdfs:range :Pseudonym ;
    rdfs:label "Has Pseudonym" ;
    rdfs:comment "Relates an author to a pseudonym.".

:Book a owl:Class ;
    rdfs:label "Book" ;
    rdfs:comment "A written work published by an author.".

:hasAuthor rdf:property ;
    rdfs:domain :Book ;
    rdfs:range :Author ;
    rdfs:label "Has Author" ;
    rdfs:comment "Relates a book to the author who wrote it.".

:PublishedWork rdf:type owl:Class ;
    rdfs:subClassOf [ rdf:type owl:UnionOf ;
        owl:members (:Book :Pseudonym) ] ;
    rdfs:label "Published Work" ;
    rdfs:comment "Either a book or a pseudonym published by an author.".
```
