Here is the OWL ontology design pattern in Turtle syntax:

```turtle
@prefix owl: <http://www.w3.org/2002/07/owl#>.
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#>.
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#>.
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#>.
@prefix :     <http://example.org/odp#>.

:Ontology a owl:Ontology ;
    rdfs:label "Legal Tasks Ontology" ;
    rdfs:comment "An ontology for managing legal tasks, court hierarchies, and bad check crime related regulations.".

:Court a owl:Class ;
    rdfs:label "Court" ;
    rdfs:comment "Represents a court instance.".

:DistrictCourt a owl:Class ;
    rdfs:subClassOf [ owl:Restriction [ owl:onProperty :hasAuthority ; owl:allValuesFrom :Court ] ] ;
    rdfs:label "District Court" ;
    rdfs:comment "Represents a district court instance.".

:State a owl:Class ;
    rdfs:label "State" ;
    rdfs:comment "Represents a state in which courts are located.".

:hasLocation rdf:Property ;
    rdfs:domain :Court ;
    rdfs:range :State ;
    rdfs:label "hasLocation" ;
    rdfs:comment "Indicates the state in which a court is located.".

:hasAuthority rdf:Property ;
    rdfs:domain :Court ;
    rdfs:range :Court ;
    rdfs:label "hasAuthority" ;
    rdfs:comment "Indicates the court that has authority over another court.".

:BadCheck a owl:Class ;
    rdfs:label "Bad Check" ;
    rdfs:comment "Represents a bad check transaction.".

:CheckPayment a owl:Class ;
    rdfs:subClassOf [ owl:Restriction [ owl:onProperty :hasTransactionType ; owl:allValuesFrom :BadCheck ] ] ;
    rdfs:label "Check Payment" ;
    rdfs:comment "Represents a check payment transaction.".

:hasTransactionType rdf:Property ;
    rdfs:domain :BadCheck ;
    rdfs:range xsd:string ;
    rdfs:label "hasTransactionType" ;
    rdfs:comment "Indicates the transaction type of a bad check.".

:hasTransactionStatus rdf:Property ;
    rdfs:domain :CheckPayment ;
    rdfs:range xsd:string ;
    rdfs:label "hasTransactionStatus" ;
    rdfs:comment "Indicates the transaction status of a check payment.".
```
