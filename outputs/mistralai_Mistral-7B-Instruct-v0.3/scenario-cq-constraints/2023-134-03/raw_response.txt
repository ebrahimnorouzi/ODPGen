:Policy a owl:Class ;
    rdfs:label "Policy" ;
    rdfs:comment "Represents the privacy policy associated with a piece of data.".

:maintainer rdf:property ;
    rdfs:domain :SolidActor ;

    rdfs:range :Policy ;
    rdfs:label "Policy" ;
