@prefix owl: <http://www.w3.org/2002/07/owl#>.
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>.

:Ontology a owl:Ontology ;
  rdfs:label "Name Change Scenario" ;
  rdfs:comment "This ontology models authors adopting different names to publish books due to name changes (e.g., after marriage).".

:Author a owl:Class ;
  rdfs:label "Author" ;
  rdfs:comment "A person who writes books.".

:Name a owl:Class ;
  rdfs:label "Name" ;
  rdfs:comment "A name used by an author.".

:GivenName rdf:Property ;
  rdfs:domain :Author ;
  rdfs:range xsd:string ;
  rdfs:label "Given Name" ;
  rdfs:comment "The given name of an author.".

:FamilyName rdf:Property ;
  rdfs:domain :Author ;
  rdfs:range xsd:string ;
  rdfs:label "Family Name" ;
  rdfs:comment "The family name of an author.".

:PublishedBook a owl:Class ;
  rdfs:label "Published Book" ;
  rdfs:comment "A book that has been published by an author.".

:Title rdf:Property ;
  rdfs:domain :PublishedBook ;
  rdfs:range xsd:string ;
  rdfs:label "Title" ;
  rdfs:comment "The title of a book.".

:AuthorName rdf:Property ;
  rdfs:domain :PublishedBook ;
  rdfs:range :Name ;
  rdfs:label "Author Name" ;

  rdfs:domain :Name ;

  rdfs:label "Name Change" ;
  rdfs:comment "A change in the name of an author.".

  rdfs:domain :NameChange ;
  rdfs:range :Name ;

  rdfs:comment "The old name of an author before the name change.".

:NewName rdf:Property ;
  rdfs:domain :NameChange ;
  rdfs:range :Name ;
  rdfs:label "New Name" ;
  rdfs:comment "The new name of an author after the name change.".

:HasNameChange rdf:Property ;
  rdfs:domain :Author ;
  rdfs:range :NameChange ;
  rdfs:label "Has Name Change" ;
  rdfs:comment "The name change of an author.".

:Name disjoint owl:Thing.
:NameChange disjoint owl:Thing.

- The pattern includes axioms for domain, range, and disjointness where justified.
- rdfs:label and rdfs:comment annotations are provided for every class and property.
