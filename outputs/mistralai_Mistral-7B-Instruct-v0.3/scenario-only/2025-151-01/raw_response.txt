**Turtle Ontology Design Pattern (ODP)**

```turtle
@prefix :     <http://example.org/odp#>.
@prefix owl:  <http://www.w3.org/2002/07/owl#>.
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#>.
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#>.
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#>.

:Ontology a owl:Ontology ;
  rdfs:label "ActionRequestOntology" ;
  rdfs:comment "This ontology models a formal description of an action request by an assignee on a target asset, enriched with contextual information.".

# Classes
:Assignee a owl:Class ;
  rdfs:label "Assignee" ;
  rdfs:comment "An employee who is assigned to perform an action.".

:Employee rdf:type owl:Class ;
  rdfs:subClassOf :Assignee ;
  rdfs:label "Employee" ;
  rdfs:comment "An employee of the organization.".

:TargetAsset a owl:Class ;
  rdfs:label "TargetAsset" ;
  rdfs:comment "The asset on which the action is to be performed.".

:Weekday a owl:Class ;
  rdfs:subClassOf xsd:string ;
  owl:unionOf ( ( xsd:string "Monday" ) ( xsd:string "Tuesday" ) ( xsd:string "Wednesday" ) ( xsd:string "Thursday" ) ( xsd:string "Friday" ) ) ;
  rdfs:label "Weekday" ;
  rdfs:comment "A day of the week.".

:EULocation a owl:Class ;
  rdfs:label "EULocation" ;
  rdfs:comment "A location within the European Union.".

:Action a owl:Class ;
  rdfs:label "Action" ;
  rdfs:comment "The action to be performed on the target asset.".

:ActionRequest a owl:Class ;
  rdf:type owl:FunctionalClass ;
  owl:hasProperty :assignee ;
  owl:hasProperty :targetAsset ;
  owl:hasProperty :action ;
  owl:hasProperty :requestTime ;
  owl:hasProperty :requestLocation ;
  rdfs:label "ActionRequest" ;
  rdfs:comment "A formal request for an action to be performed.".

# Properties
:assignee a owl:ObjectProperty ;
  rdfs:domain :ActionRequest ;
  rdfs:range :Assignee ;
  rdfs:label "Assignee" ;
  rdfs:comment "The assignee who is responsible for performing the action.".

:targetAsset a owl:ObjectProperty ;
  rdfs:domain :ActionRequest ;
  rdfs:range :TargetAsset ;
  rdfs:label "TargetAsset" ;
  rdfs:comment "The target asset on which the action is to be performed.".

:action a owl:ObjectProperty ;
  rdfs:domain :ActionRequest ;
  rdfs:range :Action ;
  rdfs:label "Action" ;
  rdfs:comment "The action to be performed on the target asset.".

:requestTime a owl:ObjectProperty ;
  rdfs:domain :ActionRequest ;
  rdfs:range xsd:dateTime ;
  rdfs:label "RequestTime" ;
  rdfs:comment "The time when the action request was made.".

:requestLocation a owl:ObjectProperty ;
  rdfs:domain :ActionRequest ;
  rdfs:range :EULocation ;
  rdfs:label "RequestLocation" ;
  rdfs:comment "The location from where the action request was made.".

# Axioms
:Em
