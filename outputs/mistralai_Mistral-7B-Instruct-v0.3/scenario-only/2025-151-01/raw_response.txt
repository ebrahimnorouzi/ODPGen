:ActionRequest a owl:Class ;
  rdfs:label "Action Request" ;
  rdfs:comment "A formal description of an action request by an assignee on a target asset.".

  rdfs:label "Assignee" ;
  rdfs:comment "An employee who is making an action request.".

  rdfs:label "Target Asset" ;
  rdfs:comment "The asset on which the action is requested.".

  rdfs:label "Employee" ;
  rdfs:comment "A person who is an employee of the organization.".

  rdfs:label "Weekday" ;
  rdfs:comment "A day of the week (Monday to Friday).".

  rdfs:label "EU Location" ;
  rdfs:comment "A location within the European Union.".

:RequestCapability a owl:Class ;
  rdfs:label "Request Capability" ;
  rdfs:comment "The ability to perform a specific action on the target asset.".

  rdfs:range :EULocation ;
  rdfs:label "Is EU Location" ;
