:Ontology a owl:Ontology ;
  rdfs:label "Author Name Changes Ontology" ;
  rdfs:comment "An ontology to model authors adopting different names to publish books.".

# Atomic Requirements
:Author a owl:Class ;
  rdfs:label "Author" ;
  rdfs:comment "A person who writes books.".

:Pseudonym a owl:Class ;
  rdfs:label "Pseudonym" ;
  rdfs:comment "An alternative name used by an author.".

:Book a owl:Class ;
  rdfs:label "Book" ;
  rdfs:comment "A written work published by an author.".

:PublishedBy a owl:ObjectProperty ;
  rdfs:domain :Author ;
  rdfs:range :Book ;
  rdfs:label "Published By" ;
  rdfs:comment "Relates an author to the books they have published.".

:UsedPseudonym a owl:ObjectProperty ;
  rdfs:domain :Author ;
  rdfs:range :Pseudonym ;
  rdfs:label "Used Pseudonym" ;
  rdfs:comment "Relates an author to the pseudonyms they have used.".

# Axioms
:Author rdfs:subClassOf [ a owl:Restriction ;
                             owl:onProperty :hasName ;
                             owl:hasValue "_" ;
                             owl:cardinality "1" ].

:Pseudonym rdfs:subClassOf [ a owl:Restriction ;
                               owl:onProperty :hasName ;
                               owl:hasValue "_" ;
                               owl:cardinality "1" ].

:Book rdfs:subClassOf [ a owl:Restriction ;
                           owl:onProperty :hasTitle ;
                           owl:hasValue "_" ;
                           owl:cardinality "1" ].

:Author owl:disjointWith :Pseudonym.
