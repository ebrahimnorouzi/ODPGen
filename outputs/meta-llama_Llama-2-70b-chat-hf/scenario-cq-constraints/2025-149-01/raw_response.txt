@prefix owl:  <http://www.w3.org/2002/07/owl#>.
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#>.
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#>.
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#>.
@prefix :     <http://example.org/odp#>.

:Ontology a owl:Ontology ;
  rdfs:label "Relational Database Ontology" ;
  rdfs:comment "Represents a relational database and its schema, tables, columns, and relationships.".

:Database a owl:Class ;
  rdfs:label "Database" ;
  rdfs:comment "Represents a relational database.".

:Table a owl:Class ;
  rdfs:label "Table" ;
  rdfs:comment "Represents a table in a relational database." ;
  owl:hasPart :Column.

:Column a owl:Class ;
  rdfs:label "Column" ;
  rdfs:comment "Represents a column in a table." ;
  owl:hasPart :Cell.

:Cell a owl:Class ;
  rdfs:label "Cell" ;
  rdfs:comment "Represents a cell in a table.".

:PrimaryKey a owl:ObjectProperty ;
  rdfs:label "PrimaryKey" ;
  rdfs:comment "Represents a primary key relationship between two tables." ;
  owl:domain :Table ;
  owl:range :Table.

:ForeignKey a owl:ObjectProperty ;
  rdfs:label "ForeignKey" ;
  rdfs:comment "Represents a foreign key relationship between two tables." ;
  owl:domain :Table ;
  owl:range :Table.

:ViewActionScope a owl:Class ;
  rdfs:label "ViewActionScope" ;
  rdfs:comment "Represents a view action scope.".

:ProcessActionScope a owl:Class ;
  rdfs:label "ProcessActionScope" ;
  rdfs:comment "Represents a process action scope.".

:Policy a owl:Class ;
  rdfs:label "Policy" ;
  rdfs:comment "Represents a policy for a specific agent." ;
  owl:hasPart :Action.

:Action a owl:Class ;
  rdfs:label "Action" ;
  rdfs:comment "Represents an action that can be taken by an agent." ;
  owl:hasPart :ViewActionScope, :ProcessActionScope.

:ReadAction a owl:Class ;
  rdfs:label "ReadAction" ;
  rdfs:comment "Represents a read action." ;
  owl:subClassOf :Action.

:ModifyAction a owl:Class ;
  rdfs:label "ModifyAction" ;
  rdfs:comment "Represents a modify action." ;
  owl:subClassOf :Action.

:TableLevelReadProhibition a owl:Class ;
  rdfs:label "TableLevelReadProhibition" ;
  rdfs:comment "Represents a table-level read prohibition for processing." ;
  owl:subClassOf :Policy.

:TableLevelReadProhibitionForViewing a owl:Class ;
  rdfs:label "TableLevelReadProhibitionForViewing" ;
  rdfs:comment "Represents a table-level read prohibition for viewing." ;
  owl:subClassOf :Policy.

:ConditionalRowLevelReadRestriction a owl:Class ;
  rdfs:label "ConditionalRowLevelReadRestriction" ;
  rdfs:comment "Represents a conditional row-level read restriction for later enforcement." ;
  owl:subClassOf :Policy.

:SQLQuery a owl:Class ;
  rdfs:label "SQLQuery"
