  rdfs:label "Variable" ;
  rdfs:comment "A variable is a function that assigns a value to each point in a dimensional space.".

:IndependentVariable a owl:Class ;
  rdfs:label "Independent Variable" ;
  rdfs:comment "An independent variable is a variable that is not dependent on any other variable.".

:DependentVariable a owl:Class ;
  rdfs:label "Dependent Variable" ;
  rdfs:comment "A dependent variable is a variable that is dependent on one or more other variables.".

:Specialization a owl:Class ;
  rdfs:label "Specialization" ;
  rdfs:comment "A specialization is a variable that is derived from a more general variable by constraining one or more independent variables.".

:Rescaling a owl:Class ;
  rdfs:label "Rescaling" ;
  rdfs:comment "A rescaling is a transformation that changes the unit of measurement of a variable, without changing the underlying dependency structure.".

:Aggregation a owl:Class ;
  rdfs:label "Aggregation" ;
  rdfs:comment "An aggregation is a function that combines values of a variable over a set of points in a dimensional space.".

:CompositeVariable a owl:Class ;
  rdfs:label "Composite Variable" ;
  rdfs:comment "A composite variable is a variable that is composed of multiple component variables.".

:ComponentVariable a owl:Class ;
  rdfs:label "Component Variable" ;
  rdfs:comment "A component variable is a variable that is part of a composite variable.".

:Projection a owl:Class ;
  rdfs:label "Projection" ;
  rdfs:comment "A projection is a function that maps a point in a higher-dimensional space to a point in a lower-dimensional space.".

:Quantization a owl:Class ;
  rdfs:label "Quantization" ;
  rdfs:comment "A quantization is a function that maps a continuous variable to a discrete variable.".

:DiscreteVariable a owl:Class ;
  rdfs:label "Discrete Variable" ;
  rdfs:comment "A discrete variable is a variable that has a finite set of values.".

:ContinuousVariable a owl:Class ;
  rdfs:label "Continuous Variable" ;
  rdfs:comment "A continuous variable is a variable that has an infinite set of values.".

:Dataset a owl:Class ;
  rdfs:label "Dataset" ;
  rdfs:comment "A dataset is a collection of data points, each with a set of values for a set of variables.".

:DataSource a owl:Class ;
  rdfs:label "DataSource" ;
  rdfs:comment "A data source is a source of data, such as a database or a file.".

:Process a owl:Class ;
  rdfs:label "Process" ;
  rdfs:comment "A process is a set of steps that transforms input data into output data.".

:Serialization a owl:Class ;
  rdfs:label "Serialization" ;
  rdfs:comment "A serialization is a representation of data in a format that can be stored or transmitted.".

:Format a owl:Class ;
  rdfs:label "Format" ;
  rdfs:comment "A format is a specific way of representing data, such as CSV or Parquet.".

:URL a owl:Class ;

:DataTransformation a owl:Class ;
  rdfs:label "DataTransformation" ;
  rdfs:comment "A data transformation is a process that transforms data from one format to another.".

:RawCaseReportDataset a owl:Class ;

:AggregatedDataset a owl:Class ;
  rdfs:label "Aggregated Dataset" ;
  rdfs:comment "An aggregated dataset is a dataset that contains data that has been aggregated from a set of input datasets.".

:Equivalence a owl:Class ;
  rdfs:label "Equivalence" ;
  rdfs:comment "An equivalence is a relationship between two variables that have the same value for a given set of points.".
