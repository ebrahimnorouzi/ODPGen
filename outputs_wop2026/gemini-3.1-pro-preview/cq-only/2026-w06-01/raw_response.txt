```turtle
@prefix owl:  <http://www.w3.org/2002/07/owl#> .

@prefix saref: <https://saref.etsi.org/core/> .
@prefix :     <http://example.org/odp#> .

:Ontology a owl:Ontology ;
    rdfs:label "PIIRRITE Ontology Design Pattern" ;
    rdfs:comment "An ontology design pattern representing the PIIRRITE navigable environment, spatial entities, contextual properties, and user requirements." .

### Classes ###

saref:Property a owl:Class ;
    rdfs:label "Property" ;
    rdfs:comment "A property as defined by the SAREF ontology." .

saref:PropertyValue a owl:Class ;
    rdfs:label "Property Value" ;
    rdfs:comment "A value associated with a SAREF property." ;
    rdfs:subClassOf 
        [ a owl:Restriction ;
          owl:onProperty :isLinkedToProperty ;
          owl:someValuesFrom saref:Property ] ,
        [ a owl:Restriction ;
          owl:onProperty :hasTemporalValidityInterval ;
          owl:cardinality "1"^^xsd:nonNegativeInteger ] .

:SpatialEntity a owl:Class ;
    rdfs:label "Spatial Entity" ;
    rdfs:comment "A spatial entity within the navigable environment." .

:NavigableEnvironment a owl:Class ;
    rdfs:label "Navigable Environment" ;
    rdfs:comment "The overall environment that can be navigated by users." .

:NavigablePath a owl:Class ;
    rdfs:label "Navigable Path" ;
    rdfs:comment "A path that can be navigated." ;
    rdfs:subClassOf :SpatialEntity ,
        [ a owl:Restriction ;
          owl:onProperty :associatedWithPOI ;
          owl:someValuesFrom :PointOfInterest ] .

:NavigablePolyline a owl:Class ;
    rdfs:label "Navigable Polyline" ;
    rdfs:comment "A navigable path represented as a polyline." ;
    rdfs:subClassOf :NavigablePath .

:PointOfInterest a owl:Class ;
    rdfs:label "Point of Interest" ;

    rdfs:subClassOf :SpatialEntity .

    rdfs:label "User Position" ;
    rdfs:comment "The spatial position of a user." .

    rdfs:label "Campus" ;
    rdfs:comment "A university or institutional campus." .

:GeometricType a owl:Class ;

    rdfs:comment "The geometric representation type of a spatial entity." .

:ContextualProperty a owl:Class ;
    rdfs:label "Contextual Property" ;
    rdfs:comment "A property providing context to a spatial entity." ;
    rdfs:subClassOf saref:Property .

:CategoricalProperty a owl:Class ;
    rdfs:label "Categorical Property" ;
    rdfs:comment "A contextual property that takes qualitative or categorical values." ;
    rdfs:subClassOf :ContextualProperty .

:AdmissibleValue a owl:Class ;
    rdfs:label "Admissible Value" ;
    rdfs:comment "An allowed value for a categorical property." .

:VendingMachine a owl:Class ;

    rdfs:subClassOf :PointOfInterest .

:BicycleParking a owl:Class ;
    rdfs:label "Bicycle Parking" ;
    rdfs:comment "A designated area for parking bicycles." ;
    rdfs:subClassOf :PointOfInterest .

:TemporalValidityConstraint a owl:Class ;
    rdfs:label "Temporal Validity Constraint" ;
    rdfs:comment "A constraint defining when a property or value is valid." .

:TemporalValidityInterval a owl:Class ;
    rdfs:label "Temporal Validity Interval" ;
    rdfs:comment "A specific time interval during which a value is valid." .

:RainfallProperty a owl:Class ;
    rdfs:label "Rainfall Property" ;
    rdfs:comment "A contextual property representing rainfall." ;
    rdfs:subClassOf :ContextualProperty .

:PollenLevelProperty a owl:Class ;
    rdfs:label "Pollen Level Property" ;
    rdfs:comment "A contextual property representing pollen levels." ;
    rdfs:subClassOf :ContextualProperty .

:Requirement a owl:Class ;
    rdfs:label "Requirement" ;
    rdfs:comment "A requirement specified by a user." .

:TargetProperty a owl:Class ;
    rdfs:label "Target Property" ;
    rdfs:comment "The property targeted by a requirement." .

:ExpectedValue a owl:Class ;
    rdfs:label "Expected Value" ;
    rdfs:comment "The value expected by a user for a target property." .

:UserProfile a owl:Class ;
    rdfs:label "User Profile" ;
    rdfs:comment "A profile representing a user and their requirements." ;
    rdfs:subClassOf 
        [ a owl:Restriction ;
          owl:onProperty :hasRequirement ;
          owl:someValuesFrom :Requirement ] .

:Condition a owl:Class ;
    rdfs:label "Condition" ;
    rdfs:comment "A condition that makes up a requirement." .

:ComposedCondition a owl:Class ;
    rdfs:label "Composed Condition" ;

:DistanceRequisite a owl:Class ;
    rdfs:label "Distance Requisite" ;
    rdfs:comment "A requisite based on distance." .

:DistanceType a owl:Class ;
    rdfs:label "Distance Type" ;
    rdfs:comment "The type of distance measurement." .

:ComparisonOperator a owl:Class ;

    rdfs:comment "A factor related to distance used in requirements." .
```
