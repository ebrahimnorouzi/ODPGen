:Ontology a owl:Ontology ;
  rdfs:label "Reverse Engineering Symbol and Behavior ODP" ;
  rdfs:comment "A minimal ontology design pattern for representing executable files, symbols, functions, variables, calls, control-flow and behavioral indicators derived from reverse engineering artifacts such as symbol tables, to support analysis of suspicious and malicious patterns." .

### Classes

:ExecutableFile a owl:Class ;
  rdfs:label "Executable file" ;
  rdfs:comment "A binary executable artifact being analyzed." .

:ProgramElement a owl:Class ;
  rdfs:label "Program element" ;
  rdfs:comment "A generic element identified in an executable during reverse engineering." .

  rdfs:label "Symbol" ;

:Function a owl:Class ;

  rdfs:comment "A function identified in an executable." ;
  rdfs:subClassOf :Symbol .

:Variable a owl:Class ;

:ImportedFunction a owl:Class ;
