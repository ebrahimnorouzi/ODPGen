```turtle
@prefix owl:  <http://www.w3.org/2002/07/owl#> .
@prefix rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .
@prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#> .
@prefix xsd:  <http://www.w3.org/2001/XMLSchema#> .
@prefix :     <http://example.org/odp#> .

:Ontology a owl:Ontology ;
  rdfs:label "Variable, dimensional space, aggregation, quantization, and dataset ODP" ;
  rdfs:comment "An ontology design pattern for describing variables, their dimensional spaces and dependencies, specialization and rescaling relations, composite variables, aggregations and candidate quantizations, dimensional-space continuity and boundedness, dataset coverage and serialization, generation processes, and asserted variable equivalence." .

#################################################################
# Classes
#################################################################

:Variable a owl:Class ;
  rdfs:label "Variable" ;
  rdfs:comment "A variable whose values are taken over a dimensional space and that may depend on independent variables." .

:IndependentVariable a owl:Class ;
  rdfs:label "Independent variable" ;
  rdfs:comment "A variable that serves as an argument or axis for another variable and is not modeled here as depending on other variables for the queried relation." ;
  rdfs:subClassOf :Variable .

:DependentVariable a owl:Class ;
  rdfs:label "Dependent variable" ;
  rdfs:comment "A variable whose values depend on one or more independent variables." ;
  rdfs:subClassOf :Variable .

:CompositeVariable a owl:Class ;
  rdfs:label "Composite variable" ;
  rdfs:comment "A variable composed of component variables." ;
  rdfs:subClassOf :Variable .

:DerivedVariable a owl:Class ;
  rdfs:label "Derived variable" ;
  rdfs:comment "A variable derived from a more general variable." ;
  rdfs:subClassOf :Variable .

:RescaledVariable a owl:Class ;
  rdfs:label "Rescaled variable" ;
  rdfs:comment "A variable obtained by rescaling another variable without changing the underlying dependency structure." ;
  rdfs:subClassOf :DerivedVariable .

:AggregatingVariable a owl:Class ;
  rdfs:label "Aggregating variable" ;
  rdfs:comment "A dependent variable that defines one or more aggregations over another variable." ;
  rdfs:subClassOf :DependentVariable .

:DimensionalSpace a owl:Class ;
  rdfs:label "Dimensional space" ;
  rdfs:comment "A space on which a variable takes its values or whose dimensions structure a variable." .

:ContinuousDimensionalSpace a owl:Class ;
  rdfs:label "Continuous dimensional space" ;
  rdfs:comment "A dimensional space modeled as continuous." ;
  rdfs:subClassOf :DimensionalSpace .

:DiscreteDimensionalSpace a owl:Class ;
  rdfs:label "Discrete dimensional space" ;
  rdfs:comment "A dimensional space modeled as discrete." ;
  rdfs:subClassOf :DimensionalSpace .

:BoundedDimensionalSpace a owl:Class ;
  rdfs:label "Bounded dimensional space" ;
  rdfs:comment "A dimensional space with explicit bounds." ;
  rdfs:subClassOf :DimensionalSpace .

:FiniteDimensionalSpace a owl:Class ;
  rdfs:label "Finite dimensional space" ;
  rdfs:comment "A discrete and bounded dimensional space that is fully enumerable with finite resources." ;
  rdfs:subClassOf :DiscreteDimensionalSpace, :BoundedDimensionalSpace .

:Quantization a owl:Class ;
  rdfs:label "Quantization" ;
  rdfs:comment "A discretization of a continuum, possibly with a stated resolution." .

:SamplingQuantization a owl:Class ;
  rdfs:label "Sampling quantization" ;
  rdfs:comment "A quantization obtained by sampling a continuum." ;
  rdfs:subClassOf :Quantization .

:BinningQuantization a owl:Class ;
  rdfs:label "Binning quantization" ;
  rdfs:comment "A quantization obtained by binning a continuum." ;
  rdfs:subClassOf :Quantization .

:PeriodicBinningQuantization a owl:Class ;
  rdfs:label "Periodic binning quantization" ;
  rdfs:comment "A binning quantization with a repeating period and an in-period resolution." ;
  rdfs:subClassOf :BinningQuantization .

:Aggregation a owl:Class ;
  rdfs:label "Aggregation" ;
  rdfs:comment "An aggregation specification that groups values of a variable along an axis using an aggregation function and may suggest candidate quantizations." .

:AggregationFunction a owl:Class ;
  rdfs:label "Aggregation function" ;
  rdfs:comment "A function used to aggregate grouped values, such as average, sum, minimum, or maximum." .

:BoundingRegion a owl:Class ;
  rdfs:label "Bounding region" ;
  rdfs:comment "A region or interval that bounds a dimensional space or dataset coverage, exactly or approximately." .

:Dataset a owl:Class ;
  rdfs:label "Dataset" ;
  rdfs:comment "A dataset that may realize variables, have coverage, be serialized by data sources, and be generated by a process." .

:DataSource a owl:Class ;
  rdfs:label "Data source" ;
  rdfs:comment "A source that offers a serialization of a dataset." .

:DataFormat a owl:Class ;
  rdfs:label "Data format" ;
  rdfs:comment "A serialization format in which a dataset is offered." .

:Serialization a owl:Class ;
  rdfs:label "Serialization" ;
  rdfs:comment "A concrete offering of a dataset by a data source in a particular data format, optionally at a URL." .

:Process a owl:Class ;
  rdfs:label "Process" ;
  rdfs:comment "A process that generates a dataset and may consume other datasets as input." .

#################################################################
# Object Properties
#################################################################

:hasDimensionalSpace a owl:ObjectProperty ;
  rdfs:label "has dimensional space" ;
  rdfs:comment "Relates a variable to the dimensional space on which it takes values or is structured." ;
  rdfs:domain :Variable ;
  rdfs:range :DimensionalSpace .

:dependsOnIndependentVariable a owl:ObjectProperty ;
  rdfs:label "depends on independent variable" ;
  rdfs:comment "Relates a dependent variable to an independent variable on which it depends." ;
  rdfs:domain :DependentVariable ;
  rdfs:range :IndependentVariable .

:isDerivedFromVariable a owl:ObjectProperty ;
  rdfs:label "is derived from variable" ;
  rdfs:comment "Relates a derived variable to the more general variable from which it is derived." ;
  rdfs:domain :DerivedVariable ;
  rdfs:range :Variable .

:constrainsIndependentVariable a owl:ObjectProperty ;
  rdfs:label "constrains independent variable" ;
  rdfs:comment "Relates a specialization variable to the independent variable that it fixes or constrains to a specific value or region." ;
  rdfs:domain :DerivedVariable ;
  rdfs:range :IndependentVariable .

:isRescaledFromVariable a owl:ObjectProperty ;
  rdfs:label "is rescaled from variable" ;
  rdfs:comment "Relates a rescaled variable to the original non-rescaled variable." ;
  rdfs:domain :RescaledVariable ;
  rdfs:range :Variable .

:hasComponentVariable a owl:ObjectProperty ;
  rdfs:label "has component variable" ;
  rdfs:comment "Relates a composite variable to one of its component variables." ;
  rdfs:domain :CompositeVariable ;
  rdfs:range :Variable .

:isProjectionOfDimensionalSpace a owl:ObjectProperty ;
  rdfs:label "is projection of dimensional space" ;
  rdfs:comment "Relates a sub-dimensional space to a composite dimensional space of which it is a projection." ;
  rdfs:domain :DimensionalSpace ;
  rdfs:range :DimensionalSpace .

:definesAggregation a owl:ObjectProperty ;
  rdfs:label "defines aggregation" ;
  rdfs:comment "Relates an aggregating variable to an aggregation that it defines." ;
  rdfs:domain :AggregatingVariable ;
  rdfs:range :Aggregation .

:aggregatesVariable a owl:ObjectProperty ;
  rdfs:label "aggregates variable" ;
  rdfs:comment "Relates an aggregation to the variable whose values are aggregated." ;
  rdfs:domain :Aggregation ;
  rdfs:range :Variable .

:groupsByIndependentVariable a owl:ObjectProperty ;
  rdfs:label "groups by independent variable" ;
  rdfs:comment "Relates an aggregation to the independent variable along which values are grouped." ;
  rdfs:domain :Aggregation ;
  rdfs:range :IndependentVariable .

:suggestsQuantization a owl:ObjectProperty ;
  rdfs:label "suggests quantization" ;
  rdfs:comment "Relates an aggregation to a candidate quantization suggested for grouping along an axis." ;
  rdfs:domain :Aggregation ;
  rdfs:range :Quantization .

:usesAggregationFunction a owl:ObjectProperty ;
  rdfs:label "uses aggregation function" ;
  rdfs:comment "Relates an aggregation to the aggregation function it uses." ;
  rdfs:domain :Aggregation ;
  rdfs:range :AggregationFunction .

:discretizesContinuum a owl:ObjectProperty ;
  rdfs:label "discretizes continuum" ;
  rdfs:comment "Relates a quantization to the continuous dimensional space that it discretizes." ;
  rdfs:domain :Quantization ;
  rdfs:range :ContinuousDimensionalSpace .

:hasBoundingRegion a owl:ObjectProperty ;
  rdfs:label "has bounding region" ;
  rdfs:comment "Relates a dimensional space or dataset to a bounding region describing its exact or approximate bounds." ;
  rdfs:range :BoundingRegion .

:hasCoverageDimensionalSpace a owl:ObjectProperty ;
  rdfs:label "has coverage dimensional space" ;
  rdfs:comment "Relates a dataset to a dimensional space describing its coverage." ;
  rdfs:domain :Dataset ;
  rdfs:range :DimensionalSpace .

:hasSerialization a owl:ObjectProperty ;
  rdfs:label "has serialization" ;
  rdfs:comment "Relates a dataset to a serialization through which it is offered." ;
  rdfs:domain :Dataset ;
  rdfs:range :Serialization .

:offeredByDataSource a owl:ObjectProperty ;
  rdfs:label "offered by data source" ;
  rdfs:comment "Relates a serialization to the data source that offers it." ;
  rdfs:domain :Serialization ;
  rdfs:range :DataSource .

:hasDataFormat a owl:ObjectProperty ;
  rdfs:label "has data format" ;
  rdfs:comment "Relates a serialization to the data format in which the dataset is offered." ;
  rdfs:domain :Serialization ;
  rdfs:range :DataFormat .

:generatedByProcess a owl:ObjectProperty ;
  rdfs:label "generated by process" ;
  rdfs:comment "Relates a dataset to the process by which it was generated." ;
  rdfs:domain :Dataset ;
  rdfs:range :Process .

:consumesInputDataset a owl:ObjectProperty ;
  rdfs:label "consumes input dataset" ;
  rdfs:comment "Relates a process to a dataset that it consumes as input." ;
  rdfs:domain :Process ;
  rdfs:range :Dataset .

:assertedEquivalentToVariable a owl:ObjectProperty ;
  rdfs:label "asserted equivalent to variable" ;
  rdfs:comment "Relates two independently modeled variables that are asserted to always coincide in value." ;
  rdfs:domain :Variable ;
  rdfs:range :Variable ;
  a owl:SymmetricProperty .

#################################################################
# Datatype Properties
#################################################################

:hasResolutionLiteral a owl:DatatypeProperty ;
  rdfs:label "has resolution literal" ;
  rdfs:comment "Gives the exact or stated resolution of a quantization, such as a temporal step or spatial bin size." ;
  rdfs:domain :Quantization ;
  rdfs:range xsd:string .

:hasPeriodLiteral a owl:DatatypeProperty ;
  rdfs:label "has period literal" ;
  rdfs:comment "Gives the repeating period of a periodic binning quantization." ;
  rdfs:domain :PeriodicBinningQuantization ;
  rdfs:range xsd:string .

:hasInPeriodResolutionLiteral a owl:DatatypeProperty ;
  rdfs:label "has in-period resolution literal" ;
  rdfs:comment "Gives the resolution within each period of a periodic binning quantization." ;
  rdfs:domain :PeriodicBinningQuantization ;
  rdfs:range xsd:string .

:isRegularBinning a owl:DatatypeProperty ;
  rdfs:label "is regular binning" ;
  rdfs:comment "States whether a binning quantization uses regular bins." ;
  rdfs:domain :BinningQuantization ;
  rdfs:range xsd:boolean .

:isApproximateBoundingRegion a owl:DatatypeProperty ;
  rdfs:label "is approximate bounding region" ;
  rdfs:comment "States whether a bounding region is approximate rather than exact." ;
  rdfs:domain :BoundingRegion ;
  rdfs:range xsd:boolean .

:hasURL a owl:DatatypeProperty ;
  rdfs:label "has URL" ;
  rdfs:comment "Gives the URL at which a serialization is available." ;
  rdfs:domain :Serialization ;
  rdfs:range xsd:anyURI .
```
